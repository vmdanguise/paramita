PRAGMA foreign_keys = ON;

-- 1. Prácticas base: oraciones, mantras, meditaciones guiadas y pausas de freno/calma
CREATE TABLE practices (
    id TEXT PRIMARY KEY NOT NULL,
    type TEXT NOT NULL CHECK (type IN ('PRAYER', 'MANTRA', 'GUIDED_MEDITATION', 'IMPULSE_PAUSE')),
    title TEXT NOT NULL,
    subtitle TEXT,
    tradition TEXT NOT NULL DEFAULT 'Budismo Tibetano · Sakya (Paramita)',
    estimated_minutes INTEGER NOT NULL DEFAULT 5,
    audio_asset_uri TEXT,
    tts_enabled INTEGER NOT NULL DEFAULT 1 CHECK (tts_enabled IN (0, 1)),
    default_pace_wpm INTEGER NOT NULL DEFAULT 120,
    merit_dedication TEXT NOT NULL,
    paramita_section TEXT,
    paramita_topic TEXT,
    paramita_url_path TEXT,
    is_favorite INTEGER NOT NULL DEFAULT 0 CHECK (is_favorite IN (0, 1)),
    updated_at_epoch_ms INTEGER NOT NULL
);

-- 2. Segmentos o fases secuenciales de cada oración, meditación o pausa
CREATE TABLE practice_segments (
    id TEXT PRIMARY KEY NOT NULL,
    practice_id TEXT NOT NULL,
    segment_order INTEGER NOT NULL,
    phase TEXT NOT NULL CHECK (
        phase IN ('MOTIVATION', 'POSTURE_BREATH', 'CORE', 'CONTEMPLATION', 'PAUSE', 'DEDICATION')

);

CREATE INDEX idx_segments_practice_order ON practice_segments(practice_id, segment_order);

-- 3. Configuración específica para recitación de mantras con repeticiones
CREATE TABLE mantra_configs (

CREATE TABLE practice_tags (

4. Decisiones clave que evitan problemas futuros
Separar practice_segments de practices: permite que una oración tenga 1 bloque, una meditación guiada tenga 5 fases con distintos tiempos de pausa, y un freno de compra tenga 2 pasos breves, usando exactamente la misma estructura.

Soporte nativo para "prenderse y recitar solo": combinar audio_asset_uri (grabaciones de referencia o campanas) con tts_enabled, default_pace_wpm, pause_after_sec y mantra_config (target_repetitions, bell_every_n, SPOKEN / WHISPER / MENTAL) permite automatizar una sesión continua sin tocar la pantalla.

Registro no punitivo en merit_and_calm_events: registrar cuándo frenaste una compra impulsiva o hiciste una pausa digital junto con la emoción notada y la dedicación del mérito refuerza la lucidez sin convertir la espiritualidad en un contador competitivo de "rachas".

Trazabilidad hacia paramita.org: paramita_section, paramita_topic y paramita_url_path mantienen ordenada la referencia a la fuente sin acoplar la app a scraping frágil ni alterar el texto litúrgico.

Diseñemos el motor en Kotlin para ejecutar segmentos, repeticiones de mantras, TTS y pausas en segundo plano en Android.

Vamos a diseñar el motor de ejecución de prácticas en Kotlin para Android, pensado para que la app se prenda, avance sola por los segmentos, repita mantras con pausas y campanas, hable con TTS o reproduzca audios y siga corriendo aunque se apague o bloquee la pantalla, manteniendo limpio el historial de sesiones y la dedicación del mérito.

1. Arquitectura del motor (separación de responsabilidades)
Para que la lógica no quede atrapada dentro de una Activity ni de un Composable:

PracticeRunnerEngine (clase de dominio / estado reactivo): máquina de estados basada en StateFlow que recorre los PracticeSegment y las repeticiones de cada mantra (1..N), gestiona los tiempos de pausa, dispara campanas/señales y expone RunnerSnapshot (segmento actual, repetición actual/objetivo, fase, modo de recitación, tiempo restante de pausa, estado PLAYING | PAUSED | COMPLETED).

SpeechAndBellController: abstrae android.speech.tts.TextToSpeech (con UtteranceProgressListener convertido a suspend mediante suspendCancellableCoroutine), ajuste de velocidad (speechRate derivado de defaultPaceWpm), y sonidos breves (como campana tibetana/cuenco al completar ciclos o cerrar).

PracticePlaybackService (ForegroundService / MediaSessionService): mantiene vivo el bucle de ejecución y la notificación persistente mientras hay una sesión activa, permitiendo pausar, reanudar o detener desde la notificación o la pantalla de bloqueo.

PracticeSessionRepository: persiste en Room (practice_sessions) el inicio, las repeticiones completadas, si se completó la sesión y si se dedicó el mérito al cerrar.

import android.content.Context
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import kotlinx.coroutines.suspendCancellableCoroutine
import java.util.Locale
import java.util.UUID
import kotlin.coroutines.resume

class SpeechAndBellController(
    context: Context,
    private val onBellCue: (isCycleMilestone: Boolean) -> Unit = {}
) {
    private val appContext = context.applicationContext
    private var isReady = false
    private var pendingCallback: (() -> Unit)? = null

    private val tts: TextToSpeech = TextToSpeech(appContext) { status ->
        if (status == TextToSpeech.SUCCESS) {
            val esAr = Locale("es", "AR")
            val esEs = Locale("es", "ES")
            val chosen = when {
                tts.isLanguageAvailable(esAr) >= TextToSpeech.LANG_AVAILABLE -> esAr
                tts.isLanguageAvailable(esEs) >= TextToSpeech.LANG_AVAILABLE -> esEs
                else -> Locale.getDefault()
            }
            tts.language = chosen
            isReady = true
            pendingCallback?.invoke()
            pendingCallback = null
        }
    }.apply {
        setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(utteranceId: String?) = Unit

            override fun onDone(utteranceId: String?) {
                continuations.remove(utteranceId)?.resume(Unit)
            }

            @Deprecated("Deprecated in Java")
            override fun onError(utteranceId: String?) {
                continuations.remove(utteranceId)?.resume(Unit)
            }
        })
    }

    private val continuations = mutableMapOf<String, kotlinx.coroutines.CancellableContinuation<Unit>>()

    fun setPaceFromWpm(wpm: Int) {
        // 120 wpm ~ 1.0f; acotado entre 0.72f y 1.25f para evitar lectura robótica o apresurada
        val rate = (wpm / 120f).coerceIn(0.72f, 1.25f)
        tts.setSpeechRate(rate)
    }

    suspend fun speakAndAwait(text: String, mode: RecitationMode): Unit {
        if (text.isBlank()) return
        if (mode == RecitationMode.MENTAL) {
            // En recitación mental no se pronuncia en voz alta; se respeta el tiempo de contemplación
            return
        }

        if (!isReady) {
            suspendCancellableCoroutine<Unit> { cont ->
                pendingCallback = { cont.resume(Unit) }
                cont.invokeOnCancellation { pendingCallback = null }
            }
        }

        val pitch = when (mode) {
            RecitationMode.SPOKEN -> 1.0f
            RecitationMode.WHISPER -> 0.86f
            RecitationMode.MENTAL -> 1.0f
        }
        tts.setPitch(pitch)

        val utteranceId = UUID.randomUUID().toString()
        suspendCancellableCoroutine<Unit> { cont ->
            continuations[utteranceId] = cont
            val result = tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, utteranceId)
            if (result == TextToSpeech.ERROR) {
                continuations.remove(utteranceId)
                if (cont.isActive) cont.resume(Unit)
            }
            cont.invokeOnCancellation {
                continuations.remove(utteranceId)
                tts.stop()
            }
        }
    }

    fun ringBell(isCycleMilestone: Boolean) {
        onBellCue(isCycleMilestone)
    }

    fun stopSpeaking() {
        tts.stop()
        continuations.values.forEach { if (it.isActive) it.resume(Unit) }
        continuations.clear()
    }

    fun shutdown() {
        stopSpeaking()
        tts.shutdown()
    }
}
4. Motor principal (PracticeRunnerEngine)
El motor maneja dos flujos complementarios:

Secuencia de segmentos y repeticiones internas (ideal para meditaciones guiadas por fases, oraciones con repetición o pausas de freno al impulso).

Bucle de repeticiones de mantra (1..targetRepetitions), donde en cada vuelta se recita el mantra (o se guarda silencio contemplativo en MENTAL), se aplica pauseBetweenRepsMs, se toca campana cada bellEveryN repeticiones y se actualiza el contador en tiempo real.

Kotlin
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.isActive
import kotlinx.coroutines.launch

class PracticeRunnerEngine(
    private val scope: CoroutineScope,
    private val speechController: SpeechAndBellController,
    private val onSessionFinished: (spec: PracticeRunSpec, completedRepetitions: Int, completedFull: Boolean) -> Unit
) {
    private val _snapshot = MutableStateFlow(RunnerSnapshot())
    val snapshot: StateFlow<RunnerSnapshot> = _snapshot.asStateFlow()

    private var runnerJob: Job? = null
    private var currentSpec: PracticeRunSpec? = null

    fun start(spec: PracticeRunSpec, initialMode: RecitationMode = RecitationMode.SPOKEN) {
        stop(recordAsCompleted = false)
        currentSpec = spec
        speechController.setPaceFromWpm(spec.defaultPaceWpm)

        val targetReps = spec.mantraConfig?.targetRepetitions ?: 0
        _snapshot.value = RunnerSnapshot(
            status = RunnerStatus.PLAYING,
            practiceId = spec.practiceId,
            practiceTitle = spec.title,
            segmentIndex = 0,
            totalSegments = spec.segments.size,
            currentSegment = spec.segments.firstOrNull(),
            segmentRepetition = 1,
            segmentTotalRepetitions = spec.segments.firstOrNull()?.repeatCount ?: 1,
            mantraRepetition = 0,
            mantraTargetRepetitions = targetReps,
            recitationMode = initialMode,
            pauseRemainingSec = 0,
            meritDedication = spec.meritDedication
        )

        runnerJob = scope.launch {
            executeSpec(spec)
        }
    }

    fun setRecitationMode(mode: RecitationMode) {
        _snapshot.update { it.copy(recitationMode = mode) }
        if (mode == RecitationMode.MENTAL) {
            speechController.stopSpeaking()
        }
    }

    fun pause() {
        if (_snapshot.value.status != RunnerStatus.PLAYING) return
        runnerJob?.cancel()
        speechController.stopSpeaking()
        _snapshot.update { it.copy(status = RunnerStatus.PAUSED, pauseRemainingSec = 0) }
    }

    fun resume() {
        val spec = currentSpec ?: return
        val current = _snapshot.value
        if (current.status != RunnerStatus.PAUSED) return

        _snapshot.update { it.copy(status = RunnerStatus.PLAYING) }
        runnerJob = scope.launch {
            executeSpec(spec, resumeFrom = current)
        }
    }

    fun stop(recordAsCompleted: Boolean = false) {
        runnerJob?.cancel()
        runnerJob = null
        speechController.stopSpeaking()

        val snap = _snapshot.value
        val spec = currentSpec
        if (recordAsCompleted && spec != null) {
            onSessionFinished(spec, snap.mantraRepetition, snap.status == RunnerStatus.COMPLETED)
        }

        currentSpec = null
        _snapshot.value = RunnerSnapshot(status = RunnerStatus.IDLE)
    }

    private suspend fun executeSpec(spec: PracticeRunSpec, resumeFrom: RunnerSnapshot? = null) {
        val startSegIdx = resumeFrom?.segmentIndex ?: 0

        for (segIdx in startSegIdx until spec.segments.size) {
            if (!scope.isActive) return
            val segment = spec.segments[segIdx]
            val startSegRep = if (resumeFrom != null && segIdx == startSegIdx) {
                resumeFrom.segmentRepetition.coerceIn(1, segment.repeatCount.coerceAtLeast(1))
            } else {
                1
            }

            for (segRep in startSegRep..segment.repeatCount.coerceAtLeast(1)) {
                if (!scope.isActive) return

                _snapshot.update {
                    it.copy(
                        status = RunnerStatus.PLAYING,
                        segmentIndex = segIdx,
                        totalSegments = spec.segments.size,
                        currentSegment = segment,
                        segmentRepetition = segRep,
                        segmentTotalRepetitions = segment.repeatCount.coerceAtLeast(1),
                        pauseRemainingSec = 0
                    )
                }

                val mode = _snapshot.value.recitationMode
                val textToSpeak = segment.transliteration?.takeIf { spec.mantraConfig != null } ?: segment.text
                speechController.speakAndAwait(textToSpeak, mode)

                if (segment.pauseAfterSec > 0) {
                    countdownSeconds(segment.pauseAfterSec)
                }
            }
        }

        val mantraCfg = spec.mantraConfig
        if (mantraCfg != null && spec.segments.isNotEmpty()) {
            val mantraSegment = spec.segments.last()
            val startMantraRep = (resumeFrom?.mantraRepetition ?: 0) + 1

            for (rep in startMantraRep..mantraCfg.targetRepetitions) {
                if (!scope.isActive) return
                val mode = _snapshot.value.recitationMode

                _snapshot.update {
                    it.copy(
                        status = RunnerStatus.PLAYING,
                        currentSegment = mantraSegment,
                        mantraRepetition = rep,
                        mantraTargetRepetitions = mantraCfg.targetRepetitions,
                        pauseRemainingSec = 0
                    )
                }

                val mantraText = mantraSegment.transliteration ?: mantraSegment.text
                if (mode == RecitationMode.MENTAL) {
                    // Estimación serena de lectura mental: ~2.6 segundos por vuelta corta
                    delay(2600L + mantraCfg.pauseBetweenRepsMs)
                } else {
                    speechController.speakAndAwait(mantraText, mode)
                    if (mantraCfg.pauseBetweenRepsMs > 0) {
                        delay(mantraCfg.pauseBetweenRepsMs)
                    }
                }

                if (mantraCfg.bellEveryN > 0 && rep % mantraCfg.bellEveryN == 0 && rep < mantraCfg.targetRepetitions) {
                    speechController.ringBell(isCycleMilestone = true)
                }
            }
        }

        speechController.ringBell(isCycleMilestone = false)
        _snapshot.update { it.copy(status = RunnerStatus.COMPLETED, pauseRemainingSec = 0) }
        onSessionFinished(spec, _snapshot.value.mantraRepetition, true)
    }

    private suspend fun countdownSeconds(totalSec: Int) {
        for (sec in totalSec downTo 1) {
            if (!scope.isActive) return
            _snapshot.update { it.copy(pauseRemainingSec = sec) }
            delay(1000L)
        }
        _snapshot.update { it.copy(pauseRemainingSec = 0) }
    }
}
5. Servicio en segundo plano (ForegroundService) y permisos esenciales
Para que la sesión siga corriendo con pantalla apagada o al cambiar de app en Android moderno (API 26–35+):

Permisos en AndroidManifest.xml:

android.permission.FOREGROUND_SERVICE

android.permission.FOREGROUND_SERVICE_MEDIA_PLAYBACK (requerido en Android 14+ / API 34+ cuando el servicio está tipado como mediaPlayback).

android.permission.WAKE_LOCK (si se necesita mantener CPU activa en ciclos largos sin wakelocks innecesarios).

Declaración del servicio: <service android:name=".service.PracticePlaybackService" android:foregroundServiceType="mediaPlayback" android:exported="false" />.

Notificación de sesión: mostrar una notificación discreta con el título de la práctica, el conteo actual (Repetición 7/21 o Segmento 2/3) y acciones rápidas (Pausar/Reanudar y Detener), evitando sonidos molestos al actualizar el progreso.

6. Cierre de sesión y dedicación del mérito
Al terminar (COMPLETED) o detener (stop(recordAsCompleted = true)), el motor dispara onSessionFinished y guarda un registro en practice_sessions:

practiceId, startedAtEpochMs, endedAtEpochMs, completedRepetitions, recitationMode, completedFull.

En la UI de cierre se muestra meritDedication ("Que por este mérito...") con un botón simple para marcar merit_dedicated = 1 y pasar a un estado calmo, sin ruidos ni pantallas invasivas.

Implementá PracticePlaybackService como ForegroundService con notificación de progreso y acciones de pausar/detener en Kotlin.

Para que la práctica siga corriendo aunque se bloquee la pantalla o salgas de la app, PracticePlaybackService debe combinarse con un ForegroundService (ServiceCompat.startForeground + ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK en Android 14+) y un CoroutineScope propio con SupervisorJob().

La notificación debe actualizarse de forma serena (con setOnlyAlertOnce(true) para que no haga ruido cada segundo ni cada repetición) y exponer acciones claras para Pausar/Reanudar y Detener mediante PendingIntent.

1. Declaración en AndroidManifest.xml
XML
<uses-permission android:name="android.permission.FOREGROUND_SERVICE" />
<uses-permission android:name="android.permission.FOREGROUND_SERVICE_MEDIA_PLAYBACK" />
<uses-permission android:name="android.permission.POST_NOTIFICATIONS" />

<application ...>
    <service
        android:name=".service.PracticePlaybackService"
        android:exported="false"
        android:foregroundServiceType="mediaPlayback" />
</application>
2. Implementación de PracticePlaybackService en Kotlin
El servicio encapsula el ciclo de vida de SpeechAndBellController y PracticeRunnerEngine, reacciona a intents de control (ACTION_START, ACTION_PAUSE, ACTION_RESUME, ACTION_STOP, ACTION_SET_MODE) y actualiza la notificación cada vez que cambia snapshot.

Kotlin
package com.paramita.advisor.service

import android.app.Notification

import com.paramita.advisor.MainActivity

import com.paramita.advisor.engine.PracticeRunnerEngine

import com.paramita.advisor.engine.RunnerStatus

import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers

class PracticePlaybackService : Service() {

    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)

    private lateinit var speechController: SpeechAndBellController
    private lateinit var runnerEngine: PracticeRunnerEngine
    private var snapshotCollectorJob: Job? = null

    private val _sessionState = MutableStateFlow(RunnerSnapshot())
    val sessionState: StateFlow<RunnerSnapshot> = _sessionState.asStateFlow()

    override fun onCreate() {
        super.onCreate()
        ensureNotificationChannel()

        speechController = SpeechAndBellController(
            context = this,
            onBellCue = { /* Reproducir sonido suave de campana tibetana desde res/raw */ }
        )

        runnerEngine = PracticeRunnerEngine(
            scope = serviceScope,
            speechController = speechController,
            onSessionFinished = { spec, completedReps, completedFull ->
                // Persistir sesión en Room mediante repositorio / WorkManager o scope del servicio
                serviceScope.launch {
                    persistSession(
                        practiceId = spec.practiceId,
                        completedRepetitions = completedReps,
                        completedFull = completedFull,
                        mode = _sessionState.value.recitationMode
                    )
                }
            }
        )

        snapshotCollectorJob = serviceScope.launch {
            runnerEngine.snapshot.collect { snap ->
                _sessionState.value = snap
                when (snap.status) {
                    RunnerStatus.PLAYING, RunnerStatus.PAUSED -> {
                        updateForegroundNotification(snap)
                    }
                    RunnerStatus.COMPLETED -> {
                        updateCompletedNotification(snap)
                        stopForegroundAndSelf(removeNotification = false)
                    }
                    RunnerStatus.IDLE -> Unit
                }
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> {
                val spec = intent.toPracticeRunSpec()
                val modeName = intent.getStringExtra(EXTRA_RECITATION_MODE)
                val mode = RecitationMode.entries.firstOrNull { it.name == modeName } ?: RecitationMode.SPOKEN

                if (spec != null) {
                    val initialSnap = RunnerSnapshot(
                        status = RunnerStatus.PLAYING,
                        practiceId = spec.practiceId,
                        practiceTitle = spec.title,
                        recitationMode = mode,
                        meritDedication = spec.meritDedication
                    )
                    startAsForeground(buildNotification(initialSnap))
                    runnerEngine.start(spec, mode)
                }
            }

            ACTION_PAUSE -> {
                runnerEngine.pause()
            }

            ACTION_RESUME -> {
                if (_sessionState.value.status == RunnerStatus.PAUSED) {
                    startAsForeground(buildNotification(_sessionState.value.copy(status = RunnerStatus.PLAYING)))
                    runnerEngine.resume()
                }
            }

            ACTION_STOP -> {
                runnerEngine.stop(recordAsCompleted = true)
                stopForegroundAndSelf(removeNotification = true)
            }

            ACTION_SET_MODE -> {
                val modeName = intent.getStringExtra(EXTRA_RECITATION_MODE)
                RecitationMode.entries.firstOrNull { it.name == modeName }?.let {
                    runnerEngine.setRecitationMode(it)
                }
            }
        }

        return START_NOT_STICKY
    }

    private fun startAsForeground(notification: Notification) {
        val foregroundType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK
        } else {
            0
        }
        ServiceCompat.startForeground(
            this,
            NOTIFICATION_ID,
            notification,
            foregroundType
        )
    }

    private fun updateForegroundNotification(snap: RunnerSnapshot) {
        val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        manager.notify(NOTIFICATION_ID, buildNotification(snap))
    }

    private fun updateCompletedNotification(snap: RunnerSnapshot) {
        val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val doneNotification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_lotus_calm)
            .setContentTitle(snap.practiceTitle.ifBlank { "Práctica completada" })
            .setContentText("Sesión concluida · Dedica el mérito al abrir la app")
            .setContentIntent(buildOpenAppPendingIntent())
            .setAutoCancel(true)
            .setOnlyAlertOnce(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()
        manager.notify(NOTIFICATION_ID, doneNotification)
    }

    private fun buildNotification(snap: RunnerSnapshot): Notification {
        val statusText = buildStatusSummary(snap)
        val isPaused = snap.status == RunnerStatus.PAUSED

        val toggleAction = if (isPaused) {
            NotificationCompat.Action.Builder(
                R.drawable.ic_play_calm,
                "Reanudar",
                buildServiceActionPendingIntent(ACTION_RESUME, requestCode = 101)
            ).build()
        } else {
            NotificationCompat.Action.Builder(
                R.drawable.ic_pause_calm,
                "Pausar",
                buildServiceActionPendingIntent(ACTION_PAUSE, requestCode = 102)
            ).build()
        }

        val stopAction = NotificationCompat.Action.Builder(
            R.drawable.ic_stop_calm,
            "Detener",
            buildServiceActionPendingIntent(ACTION_STOP, requestCode = 103)
        ).build()

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_lotus_calm)
            .setContentTitle(snap.practiceTitle.ifBlank { "Práctica de Dharma" })
            .setContentText(statusText)
            .setStyle(NotificationCompat.BigTextStyle().bigText(statusText))
            .setContentIntent(buildOpenAppPendingIntent())
            .setOngoing(!isPaused)
            .setOnlyAlertOnce(true)
            .setSilent(true)
            .setCategory(NotificationCompat.CATEGORY_SERVICE)
            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
            .addAction(toggleAction)
            .addAction(stopAction)
            .build()
    }

    private fun buildStatusSummary(snap: RunnerSnapshot): String {
        val parts = mutableListOf<String>()

        if (snap.status == RunnerStatus.PAUSED) {
            parts += "En pausa"
        }

        if (snap.mantraTargetRepetitions > 0) {
            parts += "Repetición ${snap.mantraRepetition}/${snap.mantraTargetRepetitions}"
        } else if (snap.totalSegments > 0) {
            val segNum = (snap.segmentIndex + 1).coerceAtMost(snap.totalSegments)
            parts += "Fase $segNum/${snap.totalSegments}"
        }

        if (snap.pauseRemainingSec > 0) {
            parts += "Pausa ${snap.pauseRemainingSec}s"
        }

        parts += snap.recitationMode.labelEs
        return parts.joinToString(" · ")
    }

    private fun buildOpenAppPendingIntent(): PendingIntent {
        val intent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }
        return PendingIntent.getActivity(
            this,
            200,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
    }

    private fun buildServiceActionPendingIntent(action: String, requestCode: Int): PendingIntent {
        val intent = Intent(this, PracticePlaybackService::class.java).apply {
            this.action = action
        }
        return PendingIntent.getService(
            this,
            requestCode,
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
    }

    private fun ensureNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Práctica y Recitación",
                NotificationManager.IMPORTANCE_LOW

                description = "Mantiene activa la recitación de oraciones, mantras y meditaciones guiadas"
                setShowBadge(false)
                enableVibration(false)
            }
            val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            manager.createNotificationChannel(channel)
        }
    }

    private fun stopForegroundAndSelf(removeNotification: Boolean) {

            if (removeNotification) ServiceCompat.STOP_FOREGROUND_REMOVE
            else ServiceCompat.STOP_FOREGROUND_DETACH

    override fun onBind(intent: Intent?): IBinder? = null

    companion object {
        const val CHANNEL_ID = "paramita_practice_channel"
        const val NOTIFICATION_ID = 4081

        const val ACTION_START = "com.paramita.advisor.action.START_PRACTICE"
        const val ACTION_PAUSE = "com.paramita.advisor.action.PAUSE_PRACTICE"
        const val ACTION_RESUME = "com.paramita.advisor.action.RESUME_PRACTICE"
        const val ACTION_STOP = "com.paramita.advisor.action.STOP_PRACTICE"
        const val ACTION_SET_MODE = "com.paramita.advisor.action.SET_RECITATION_MODE"

        const val EXTRA_PRACTICE_TITLE = "extra_practice_title"
        const val EXTRA_PACE_WPM = "extra_pace_wpm"
        const val EXTRA_MERIT_DEDICATION = "extra_merit_dedication"
        const val EXTRA_TARGET_REPS = "extra_target_reps"
        const val EXTRA_BELL_EVERY_N = "extra_bell_every_n"
        const val EXTRA_PAUSE_BETWEEN_MS = "extra_pause_between_ms"

            val intent = Intent(context, PracticePlaybackService::class.java).apply {
                action = ACTION_START
                putExtra(EXTRA_SESSION_CACHE_KEY, spec.practiceId)
                putExtra(EXTRA_PRACTICE_ID, spec.practiceId)
                putExtra(EXTRA_PRACTICE_TITLE, spec.title)
                putExtra(EXTRA_PACE_WPM, spec.defaultPaceWpm)
                putExtra(EXTRA_MERIT_DEDICATION, spec.meritDedication)
                putExtra(EXTRA_RECITATION_MODE, mode.name)
            }

        }

        fun pausePractice(context: Context) {

        fun stopPractice(context: Context) {

        fun setRecitationMode(context: Context, mode: RecitationMode) {
            context.startService(
                Intent(context, PracticePlaybackService::class.java).apply {
                    action = ACTION_SET_MODE
                    putExtra(EXTRA_RECITATION_MODE, mode.name)
                }

        private fun Intent.toPracticeRunSpec(): PracticeRunSpec? {
            val cacheKey = getStringExtra(EXTRA_SESSION_CACHE_KEY)
            if (cacheKey != null) {
                PracticeSessionCache.get(cacheKey)?.let { return it }
            }
            val id = getStringExtra(EXTRA_PRACTICE_ID) ?: return null
            val title = getStringExtra(EXTRA_PRACTICE_TITLE) ?: return null
            return PracticeRunSpec(
                practiceId = id,
                title = title,
                defaultPaceWpm = getIntExtra(EXTRA_PACE_WPM, 120),
                meritDedication = getStringExtra(EXTRA_MERIT_DEDICATION).orEmpty(),

3. Puntos clave de Android que conviene cuidar
Permiso POST_NOTIFICATIONS en Android 13+ (API 33+): si el usuario aún no concedió el permiso de notificaciones, el servicio igual puede iniciar como ForegroundService, pero la notificación no se mostrará en la barra hasta pedir el permiso en un momento sereno de la app.

No saturar NotificationManager.notify: en PracticeRunnerEngine, si hacés un countdownSeconds de 15 segundos, actualizar la notificación cada segundo está bien en meditaciones o pausas, pero mantener setOnlyAlertOnce(true) y IMPORTANCE_LOW evita vibraciones o sonidos repetitivos.

Desacoplar el guardado final: en onSessionFinished, el servicio corre dentro de serviceScope (que sigue vivo aunque el usuario cierre la UI), lo que garantiza que la sesión quede registrada aunque hayas bloqueado el teléfono y vuelto horas después.

inner class LocalBinder : Binder() {
    fun getService(): PracticePlaybackService = this@PracticePlaybackService
}

override fun onBind(intent: Intent?): IBinder = binder

@Composable
fun rememberPracticeServiceState(): State<RunnerSnapshot> {
    val context = LocalContext.current
    var serviceStateFlow by remember {
        mutableStateOf<StateFlow<RunnerSnapshot>>(MutableStateFlow(RunnerSnapshot()))
    }

    DisposableEffect(context) {
        val connection = object : ServiceConnection {
            override fun onServiceConnected(name: ComponentName?, service: IBinder?) {
                val localBinder = service as? PracticePlaybackService.LocalBinder
                localBinder?.getService()?.let { svc ->
                    serviceStateFlow = svc.sessionState
                }
            }

            override fun onServiceDisconnected(name: ComponentName?) {
                serviceStateFlow = MutableStateFlow(RunnerSnapshot())
            }
        }

        val intent = Intent(context, PracticePlaybackService::class.java)
        context.bindService(intent, connection, Context.BIND_AUTO_CREATE)

        onDispose {
            runCatching { context.unbindService(connection) }
        }
    }

    return androidx.compose.runtime.collectAsState(initial = serviceStateFlow.value)
}

@Composable
private fun ProgressStatusCard(snap: RunnerSnapshot) {
    val hasMantraTarget = snap.mantraTargetRepetitions > 0
    val progress = when {
        hasMantraTarget -> (snap.mantraRepetition.toFloat() / snap.mantraTargetRepetitions.coerceAtLeast(1))
            .coerceIn(0f, 1f)
        snap.totalSegments > 0 -> ((snap.segmentIndex + 1).toFloat() / snap.totalSegments)
            .coerceIn(0f, 1f)
        else -> 0f
    }

    Card(
        modifier = Modifier.fillMaxWidth(),
        colors = CardDefaults.cardColors(
            containerColor = MaterialTheme.colorScheme.surfaceContainerLowest
        ),
        border = BorderStroke(1.dp, MaterialTheme.colorScheme.outlineVariant)
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(14.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Text(
                    text = when (snap.status) {
                        RunnerStatus.IDLE -> "Listo para comenzar"
                        RunnerStatus.PLAYING -> "En práctica · ${snap.recitationMode.labelEs}"
                        RunnerStatus.PAUSED -> "En pausa serena"
                        RunnerStatus.COMPLETED -> "Sesión completada"
                    },
                    style = MaterialTheme.typography.labelLarge,
                    color = MaterialTheme.colorScheme.onSurface
                )

                Text(
                    text = if (hasMantraTarget) {
                        "${snap.mantraRepetition} / ${snap.mantraTargetRepetitions}"
                    } else if (snap.totalSegments > 0) {
                        "Fase ${(snap.segmentIndex + 1).coerceAtMost(snap.totalSegments)} / ${snap.totalSegments}"
                    } else {
                        ""
                    },
                    style = MaterialTheme.typography.labelLarge,
                    color = MaterialTheme.colorScheme.primary
                )
            }

            LinearProgressIndicator(
                progress = { progress },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(6.dp)
            )
        }
    }
}

Cambio en caliente del modo (Voz / Susurro / Mental): al tocar un FilterChip mientras la práctica está activa, llama a PracticePlaybackService.setRecitationMode(context, mode), así que en la siguiente repetición o segmento el motor ya aplica el nuevo modo sin cortar la sesión.

Diseñemos el flujo exacto en Android para detectar o frenar compras impulsivas y fomentar la calma digital con prácticas de amor bondadoso y mérito.

Vamos a diseñar el flujo exacto en Android para detectar o frenar compras impulsivas y fomentar la calma digital, integrándolo de forma natural con amor bondadoso (maitri), recitación/meditación y dedicación del mérito, sin convertir el teléfono en una cárcel ni generar culpa.

1. Principio de diseño: acompañamiento lúcido, no bloqueo punitivo
En el budismo cotidiano, el impulso de comprar por ansiedad, cansancio o vacío no se resuelve solo con "prohibirse" cosas: se aborda reconociendo la emoción y el apego momentáneo, introduciendo un espacio de calma y redirigiendo la energía hacia algo que nutra de verdad (generosidad, paz mental, descanso o práctica).

Por eso el flujo en Android combina dos vías:

Activación voluntaria instantánea (acceso rápido): botón de "Pausa / Freno de impulso" desde el inicio de la app o tile/acceso directo cuando sentís la tentación de abrir una tienda o quedarte atrapado en el scroll.

Detección suave en segundo plano (opt-in explícito del usuario):

En Android, UsageStatsManager (PACKAGE_USAGE_STATS) permite revisar ventanas recientes de uso o programar un chequeo periódico con WorkManager, mientras que un AccessibilityService (solo si el usuario lo activa explícitamente en Ajustes del sistema y se declara con transparencia su propósito) permite detectar la apertura de una app objetivo (mindful_app_targets) y lanzar una pantalla de pausa serena (ImpulsePauseActivity / ImpulsePauseSheet) respetando el cooldown_minutes para no molestar en cada toque.

3. Detalle de cada etapa y su conexión con el modelo de datos
Etapa 1: Detención suave y respiración (sin juicio)
Al abrir una app marcada en mindful_app_targets (por ejemplo, una tienda online o red de scroll continuo) o al tocar el botón de freno en la app:

Se verifica si ya pasó cooldown_minutes desde la última pausa mostrada para ese package_name (por ejemplo, 15 o 30 minutos).

Se presenta una pantalla calmada (nunca alarmas rojas ni mensajes agresivos): "Un respiro antes de avanzar. ¿Qué está buscando tu mente en este momento?".

Etapa 2: Nombrar la emoción o el impulso
El usuario elige una opción sin texto largo ni culpa:

Ansiedad / tensión

Cansancio / saturación

Aburrimiento / búsqueda de alivio

Ganas de comprar por impulso (opcionalmente puede anotar un monto estimado en ARS o dejarlo vacío).

Etapa 3: Práctica corta de amor bondadoso o calma
Según lo elegido —o por elección libre—, ofrece un atajo directo:

Amor bondadoso (maitri): cultivar amabilidad hacia uno mismo cuando hay inquietud o autocrítica ("Que esté en paz, que suelte esta tensión, que encuentre alivio genuino"), y extenderlo a quienes nos rodean.

Recitación corta: disparar PracticePlaybackService con pocas repeticiones (por ejemplo, 3 o 7 vueltas de mantra o una oración de refugio).

Soltar pantalla: cerrar con tranquilidad y dejar el teléfono boca abajo.

Etapa 4: Freno registrado como mérito y bienestar digital
Cuando el usuario decide no comprar por impulso o cerrar la sesión de scroll:

Se inserta un registro en merit_and_calm_events con:

event_type: 'IMPULSE_PURCHASE_AVOIDED' o 'DIGITAL_CALM_BREAK'

trigger_context: nombre de la app o contexto ("tienda online", "noche", etc.)

emotion_noticed: la emoción seleccionada

avoided_amount_estimate y currency_code (si se cargó)

dedication_note: dedicatoria del mérito

Se muestra la dedicatoria del mérito: por ejemplo, que la calma cultivada, el tiempo cuidado o el gasto evitado por ansiedad se dediquen al bienestar propio y al alivio de todos los seres.

class ImpulseCooldownTracker(
    private val lastShownByPackage: MutableMap<String, Long> = mutableMapOf()
) {
    fun shouldTrigger(
        packageName: String,
        cooldownMinutes: Int,
        nowEpochMs: Long = System.currentTimeMillis()
    ): Boolean {
        val lastShown = lastShownByPackage[packageName] ?: return true
        val cooldownMs = cooldownMinutes.coerceAtLeast(1) * 60_000L
        return (nowEpochMs - lastShown) >= cooldownMs
    }

    fun markShown(packageName: String, nowEpochMs: Long = System.currentTimeMillis()) {
        lastShownByPackage[packageName] = nowEpochMs
    }
}

Respiración inicial (pausa de 6 segundos opcional o inmediata): corta el automatismo antes de comprar o seguir deslizando.

Reconocer la emoción detrás del impulso: inquietud/tensión, cansancio, aburrimiento/vacío o ganas de comprar por impulso (con un campo numérico opcional en ARS si se quiere registrar el gasto frenado).

Decisión consciente y dedicatoria del mérito: guarda el evento en MeritAndCalmDao (IMPULSE_PURCHASE_AVOIDED o DIGITAL_CALM_BREAK) y cierra con la dedicatoria del mérito, manteniendo siempre la opción serena de continuar si la compra o acción es realmente necesaria.

                                displayContext = sourceAppName ?: "Pausa manual",

                                outcome = PauseOutcome.PAUSED_AND_LEFT_APP,

                                meritDedication = defaultDedication
                            )

                            val entity = MeritAndCalmEventEntity(

                                } else {

                                "Frenar compra y dedicar el mérito"
                            } else {
                                "Soltar pantalla y dedicar el mérito"
                            }

                    }

                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceBetween,
                        verticalAlignment = Alignment.CenterVertically
                    ) {

                    color = MaterialTheme.colorScheme.primaryContainer.copy(alpha = 0.45f),
                    shape = MaterialTheme.shapes.large,
                    modifier = Modifier.fillMaxWidth()

                        )

                    }

Cómo encaja con el resto de la app
Conexión directa con PracticePlaybackService: el botón "Escuchar práctica guiada de amor bondadoso / mantra" dispara PracticePlaybackService.startPractice(...) con prac_med_amor_bondadoso o prac_mantra_corazon en modo 7 repeticiones, permitiendo bloquear la pantalla y seguir escuchando.

Registro en MeritAndCalmDao: onSaveMeritAndCalmEvent llama a meritAndCalmDao.insertEvent(entity) dentro de un viewModelScope.launch, alimentando de forma sobria el historial de pausas, actos de generosidad y calma digital.

Diseñemos la sección de Buddha, Dharma y Sangha conectada a los temas y prácticas de paramita.org en Jetpack Compose.

Vamos a diseñar la sección de Buddha, Dharma y Sangha en Jetpack Compose (RefugeAndParamitaScreen), anclada en los temas, cursos, meditaciones y espacios comunitarios de paramita.org y conectada directamente con las prácticas de la base Room (PracticeDao).

1. Estructura de la sección: Tres Joyas + Ejes de Paramita
La pantalla se organiza para que el usuario pueda moverse entre el sentido profundo del refugio y las prácticas o recursos concretos de Paramita:

Selector de las Tres Joyas (BUDDHA, DHARMA, SANGHA):

Buddha: la cualidad despierta, el ejemplo de compasión y lucidez encarnado en la vida cotidiana, y la inspiración de las enseñanzas del linaje Sakya (El Buddha hoy, Libertad Interior con el Ven. Khenpo / Lama Rinchen Gyaltsen).

Dharma: la enseñanza y el método —karma como ley natural de causa y efecto, Las Cuatro Nobles Verdades, El Camino de las Paramitas, Las 37 Prácticas de los Bodhisattvas, el Sutra del Corazón y el amor bondadoso (maitri)— vinculando cada tema con prácticas listas para iniciar o recitar.

Sangha: la comunidad de práctica y estudio que sostiene el camino —encuentros diarios y semanales de meditación, recitación del Sutra del Corazón (con horarios de referencia para Latinoamérica/Argentina y España en paramita.org), cursos online y centros/círculos locales— más un botón sobrio para registrar una conexión con la Sangha (SANGHA_CONNECTION) en MeritAndCalmDao.

Acceso directo a recursos de Paramita (paramita.org / cursos.paramita.org): abre el enlace oficial en el navegador del sistema (Intent.ACTION_VIEW) sin incrustar un navegador pesado dentro de la app, cuidando el bienestar digital.

2. Mapper DTO → Entidades Room (CatalogMapper)
El mapper transforma un CatalogBundleDto en listas listas para PracticeDao.replaceCatalogBundle(...), normalizando valores por defecto y respetando existingFavorites para no borrar lo que el usuario marcó como favorito.

Kotlin
package com.paramita.advisor.data.catalog

import com.paramita.advisor.data.local.entity.MantraConfigEntity
import com.paramita.advisor.data.local.entity.PracticeEntity
import com.paramita.advisor.data.local.entity.PracticeJewelFocusEntity
import com.paramita.advisor.data.local.entity.PracticeSegmentEntity
import com.paramita.advisor.data.local.entity.PracticeTagEntity
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json

data class MappedCatalogEntities(
    val practices: List<PracticeEntity>,
    val segments: List<PracticeSegmentEntity>,
    val mantraConfigs: List<MantraConfigEntity>,
    val tags: List<PracticeTagEntity>,
    val jewels: List<PracticeJewelFocusEntity>
)

class CatalogMapper(
    private val json: Json = Json { ignoreUnknownKeys = true }
) {
    fun mapBundle(
        bundle: CatalogBundleDto,
        existingFavorites: Set<String> = emptySet(),
        nowEpochMs: Long = System.currentTimeMillis()
    ): MappedCatalogEntities {
        val practices = mutableListOf<PracticeEntity>()
        val segments = mutableListOf<PracticeSegmentEntity>()
        val mantraConfigs = mutableListOf<MantraConfigEntity>()
        val tags = mutableListOf<PracticeTagEntity>()
        val jewels = mutableListOf<PracticeJewelFocusEntity>()

        for (dto in bundle.practices) {
            val practiceId = dto.id.trim()
            if (practiceId.isEmpty()) continue

            practices += PracticeEntity(
                id = practiceId,
                type = normalizePracticeType(dto.type),
                title = dto.title.trim(),
                subtitle = dto.subtitle?.trim()?.takeIf { it.isNotEmpty() },
                tradition = dto.tradition.trim().ifEmpty { "Budismo Tibetano · Sakya (Paramita)" },
                estimatedMinutes = dto.estimatedMinutes.coerceAtLeast(1),
                audioAssetUri = dto.audioAssetUri?.trim()?.takeIf { it.isNotEmpty() },
                ttsEnabled = dto.ttsEnabled,
                defaultPaceWpm = dto.defaultPaceWpm.coerceIn(70, 190),
                meritDedication = dto.meritDedication.trim().ifEmpty {
                    "Que este mérito se dedique al bienestar y la liberación de todos los seres."
                },
                paramitaSection = dto.paramitaRef?.section?.trim(),
                paramitaTopic = dto.paramitaRef?.topic?.trim(),
                paramitaUrlPath = dto.paramitaRef?.urlPath?.trim(),
                isFavorite = practiceId in existingFavorites,
                updatedAtEpochMs = nowEpochMs
            )

            dto.segments.forEachIndexed { index, seg ->
                val order = if (seg.order > 0) seg.order else index + 1
                segments += PracticeSegmentEntity(
                    id = "${practiceId}_seg_$order",
                    practiceId = practiceId,
                    segmentOrder = order,
                    phase = normalizePhase(seg.phase),
                    heading = seg.heading.trim().ifEmpty { "Paso $order" },
                    text = seg.text.trim(),
                    transliteration = seg.transliteration?.trim()?.takeIf { it.isNotEmpty() },
                    meaning = seg.meaning?.trim()?.takeIf { it.isNotEmpty() },
                    pauseAfterSec = seg.pauseAfterSec.coerceAtLeast(0),
                    repeatCount = seg.repeatCount.coerceAtLeast(1)
                )
            }

            dto.mantraConfig?.let { m ->
                mantraConfigs += MantraConfigEntity(
                    practiceId = practiceId,
                    targetRepetitions = m.targetRepetitions.coerceAtLeast(1),
                    presetCountsJson = json.encodeToString(
                        m.presetCounts.filter { it > 0 }.ifEmpty { listOf(3, 7, 21, 108) }
                    ),
                    recitationModesJson = json.encodeToString(
                        m.recitationModes.map { it.trim().uppercase() }.ifEmpty {
                            listOf("SPOKEN", "WHISPER", "MENTAL")
                        }
                    ),
                    bellEveryN = m.bellEveryN.coerceAtLeast(0),
                    pauseBetweenRepsMs = m.pauseBetweenRepsMs.coerceAtLeast(0L)
                )
            }

            dto.tags
                .map { it.trim().lowercase() }
                .filter { it.isNotEmpty() }
                .distinct()
                .forEach { tag ->
                    tags += PracticeTagEntity(practiceId = practiceId, tag = tag)
                }

            dto.jewelFocus
                .mapNotNull { normalizeJewel(it) }
                .distinct()
                .forEach { jewel ->
                    jewels += PracticeJewelFocusEntity(practiceId = practiceId, jewel = jewel)
                }
        }

        return MappedCatalogEntities(
            practices = practices,
            segments = segments,
            mantraConfigs = mantraConfigs,
            tags = tags,
            jewels = jewels
        )
    }

    private fun normalizePracticeType(raw: String): String = when (raw.trim().uppercase()) {
        "PRAYER", "ORACION", "ORACIÓN" -> "PRAYER"
        "MANTRA" -> "MANTRA"
        "GUIDED_MEDITATION", "MEDITATION", "MEDITACION", "MEDITACIÓN" -> "GUIDED_MEDITATION"
        "IMPULSE_PAUSE", "PAUSE" -> "IMPULSE_PAUSE"
        else -> "PRAYER"
    }

    private fun normalizePhase(raw: String): String = when (raw.trim().uppercase()) {
        "MOTIVATION" -> "MOTIVATION"
        "POSTURE_BREATH", "BREATH" -> "POSTURE_BREATH"
        "CORE" -> "CORE"
        "CONTEMPLATION" -> "CONTEMPLATION"
        "PAUSE" -> "PAUSE"
        "DEDICATION" -> "DEDICATION"
        else -> "CORE"
    }

    private fun normalizeJewel(raw: String): String? = when (raw.trim().uppercase()) {
        "BUDDHA" -> "BUDDHA"
        "DHARMA", "DARMA" -> "DHARMA"
        "SANGHA" -> "SANGHA"
        else -> null
    }
}
3. Siembra inicial al arrancar la APK (CatalogSeeder)
CatalogSeeder lee assets/catalog_v1.json, guarda un indicador sencillo en SharedPreferences (seeded_schema_version) para que el arranque de la app sea instantáneo una vez sembrado, y si se sube schema_version en una futura actualización del catálogo, re-sincroniza preservando los favoritos.

Kotlin
package com.paramita.advisor.data.catalog

import android.content.Context
import com.paramita.advisor.data.local.dao.PracticeDao
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.withContext
import kotlinx.serialization.json.Json

class CatalogSeeder(
    private val context: Context,
    private val practiceDao: PracticeDao,
    private val mapper: CatalogMapper = CatalogMapper(),
    private val json: Json = Json { ignoreUnknownKeys = true },
    private val assetFileName: String = "catalog_v1.json"
) {
    private val prefs by lazy {
        context.getSharedPreferences("paramita_catalog_prefs", Context.MODE_PRIVATE)
    }

    suspend fun seedIfNeeded(forceReload: Boolean = false): Result<Int> = withContext(Dispatchers.IO) {
        runCatching {
            val rawJson = context.assets.open(assetFileName).bufferedReader(Charsets.UTF_8).use { it.readText() }
            val bundle = json.decodeFromString<CatalogBundleDto>(rawJson)

            val lastSeededVersion = prefs.getInt(KEY_SEEDED_VERSION, 0)
            val alreadyHasPractices = practiceDao.observeAllPracticesWithDetails().first().isNotEmpty()

            if (!forceReload && alreadyHasPractices && lastSeededVersion >= bundle.schemaVersion) {
                return@runCatching bundle.practices.size
            }

            val existingFavorites = practiceDao.observeAllPracticesWithDetails().first()
                .filter { it.practice.isFavorite }
                .map { it.practice.id }
                .toSet()

            val mapped = mapper.mapBundle(
                bundle = bundle,
                existingFavorites = existingFavorites
            )

            practiceDao.replaceCatalogBundle(
                practices = mapped.practices,
                segments = mapped.segments,
                mantraConfigs = mapped.mantraConfigs,
                tags = mapped.tags,
                jewels = mapped.jewels
            )

            prefs.edit().putInt(KEY_SEEDED_VERSION, bundle.schemaVersion).apply()
            mapped.practices.size
        }
    }

    companion object {
        private const val KEY_SEEDED_VERSION = "seeded_schema_version"
    }
}
Cómo llamarlo al iniciar la app
En tu Application.onCreate() (o en un Initializer / ViewModel de arranque), lanzá catalogSeeder.seedIfNeeded() dentro de un CoroutineScope con SupervisorJob() en background (Dispatchers.IO).

2. Preguntas Frecuentes (FAQ)
¿Esta app reemplaza a un maestro, un lama o los canales oficiales de Paramita?
No. La APK es una herramienta de apoyo cotidiano para organizar oraciones, contar repeticiones de mantras, acompañar meditaciones guiadas, fomentar el amor bondadoso y ayudarte a frenar impulsos en el teléfono. Las enseñanzas profundas, las dudas doctrinales, las transmisiones o el seguimiento espiritual corresponden a los canales oficiales de paramita.org y al acompañamiento de un maestro o guía calificado.

¿Por qué no hay "logros" de nivel espiritual o despertar?
Porque convertir la práctica budista en medallas de superioridad o etapas de "iluminación" alimenta el mismo apego y orgullo que la práctica busca soltar. Los hitos de la app son solo recordatorios de hábito saludable, amor bondadoso y generosidad, siempre orientados a la dedicatoria del mérito.

¿Cómo ayuda realmente a dejar de comprar por impulso o calmar el uso del celular?
No actúa como un bloqueo punitivo, sino como una pausa lúcida: te da unos segundos para respirar, notar qué emoción hay detrás del impulso (cansancio, ansiedad, aburrimiento, búsqueda de consuelo rápido), cultivar maitri y elegir conscientemente si querés soltar la pantalla o continuar con claridad.

¿Funciona sin conexión a internet?
Sí. El catálogo (catalog_v1.json), la base local con Room, el contador de repeticiones, el motor TextToSpeech en español y el ForegroundService están pensados para operar sin depender de conexión, y los enlaces a paramita.org solo se abren cuando vos decidís visitarlos.

¿Qué pasa si un día me distraigo, compro algo impulsivamente o no medito?
Nada se rompe ni se penaliza. La práctica se retoma desde donde estás, con amabilidad y sin culpa: observar el tropiezo con lucidez ya es parte del aprendizaje.

Implementá un Activity en Android que lance ImpulsePauseScreen al abrir una app configurada en mindful_app_targets usando ImpulseCooldownTracker.

Vamos a implementar ImpulsePauseActivity en Android (junto con el observador/disparador que revisa mindful_app_targets en Room y respeta ImpulseCooldownTracker) para que, cuando se abra una app marcada como objetivo (compras impulsivas o scroll continuo) y haya pasado el tiempo de enfriamiento (cooldown_minutes), se lance ImpulsePauseScreen de forma serena y sin bloquear de golpe la autonomía del usuario.

1. Cómo se dispara en Android (con opt-in explícito)
En Android moderno, una app no puede espiar silenciosamente qué abre el usuario sin un mecanismo explícito del sistema:

Vía AccessibilityService (opt-in del usuario): escucha TYPE_WINDOW_STATE_CHANGED, toma event.packageName?.toString(), ignora la propia app o el launcher del sistema, consulta mindfulAppTargetDao.getTargetByPackage(pkg) y, si promptEnabled es true y cooldownTracker.shouldTrigger(pkg, target.cooldownMinutes) devuelve true, marca markShown(pkg) y lanza ImpulsePauseActivity con FLAG_ACTIVITY_NEW_TASK y FLAG_ACTIVITY_CLEAR_TOP.

Evitar bucles infinitos: si el usuario elige "Continuar con consciencia", cooldownTracker.markShown(pkg) ya quedó registrado al abrir la pausa, por lo que al volver a la app de compras/scroll no se volverá a disparar hasta que venza cooldown_minutes.

2. Disparador de objetivos (MindfulAppTriggerEvaluator)
Kotlin
package com.paramita.advisor.digitalwellbeing

import com.paramita.advisor.data.local.dao.MindfulAppTargetDao
import com.paramita.advisor.data.local.entity.MindfulAppTargetEntity

class MindfulAppTriggerEvaluator(
    private val targetDao: MindfulAppTargetDao,
    private val cooldownTracker: ImpulseCooldownTracker
) {
    suspend fun evaluateForegroundPackage(
        packageName: String?,
        ownPackageName: String,
        nowEpochMs: Long = System.currentTimeMillis()
    ): MindfulAppTargetEntity? {
        if (packageName.isNullOrBlank() || packageName == ownPackageName) {
            return null
        }

        val target = targetDao.getTargetByPackage(packageName) ?: return null
        if (!target.promptEnabled) return null

        if (!cooldownTracker.shouldTrigger(packageName, target.cooldownMinutes, nowEpochMs)) {
            return null
        }

        cooldownTracker.markShown(packageName, nowEpochMs)
        return target
    }
}

class ImpulsePauseActivity : ComponentActivity() {

    // En producción, inyectar vía Hilt/Koin o resolver desde Application
    private val meritAndCalmDao: MeritAndCalmDao by lazy {
        (application as ParamitaApplicationHolder).meritAndCalmDao
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()

        val sourcePackageName = intent.getStringExtra(EXTRA_PACKAGE_NAME)
        val sourceAppName = intent.getStringExtra(EXTRA_APP_DISPLAY_NAME)

        setContent {
            MaterialTheme {
                ImpulsePauseScreen(
                    sourceAppName = sourceAppName,
                    sourcePackageName = sourcePackageName,
                    onSaveMeritAndCalmEvent = { event: MeritAndCalmEventEntity, _: ImpulsePauseDecision ->
                        lifecycleScope.launch {
                            meritAndCalmDao.insertEvent(event)
                        }
                    },
                    onStartGuidedMaitriPractice = {
                        // Lanzar práctica guiada de amor bondadoso o mantra en PracticePlaybackService
                        // y cerrar esta pantalla de pausa para volver a un espacio calmo
                        finish()
                    },
                    onProceedConsciously = {
                        // Si la acción es consciente y necesaria, cerramos la pausa sin juzgar
                        finish()
                    },
                    onCloseToCalm = {
                        // Llevar al usuario al home / calma en vez de dejarlo atrapado en la tienda
                        val homeIntent = packageManager.getLaunchIntentForPackage(packageName)?.apply {
                            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
                        }
                        if (homeIntent != null) {
                            startActivity(homeIntent)
                        }
                        finish()
                    }
                )
            }
        }
    }

    companion object {
        const val EXTRA_PACKAGE_NAME = "extra_source_package_name"
        const val EXTRA_APP_DISPLAY_NAME = "extra_source_app_display_name"

        fun createIntent(
            context: Context,
            packageName: String? = null,
            appDisplayName: String? = null
        ): Intent = Intent(context, ImpulsePauseActivity::class.java).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
            packageName?.let { putExtra(EXTRA_PACKAGE_NAME, it) }
            appDisplayName?.let { putExtra(EXTRA_APP_DISPLAY_NAME, it) }
        }

        fun launch(
            context: Context,
            packageName: String? = null,
            appDisplayName: String? = null
        ) {
            context.startActivity(createIntent(context, packageName, appDisplayName))
        }
    }
}

                description = "Sostener el estudio o la meditación en vínculo con la comunidad.",

fun PracticeWithDetails.toPracticeRunSpec(): PracticeRunSpec {
    val mappedSegments = orderedSegments.map { seg ->
        PracticeSegmentModel(
            id = seg.id,
            order = seg.segmentOrder,
            phase = seg.phase,
            heading = seg.heading,
            text = seg.text,
            transliteration = seg.transliteration,
            meaning = seg.meaning,
            pauseAfterSec = seg.pauseAfterSec,
            repeatCount = seg.repeatCount
        )
    }

    val mappedMantra = mantraConfig?.let { cfg ->
        MantraRunConfig(
            targetRepetitions = cfg.targetRepetitions,
            bellEveryN = cfg.bellEveryN,
            pauseBetweenRepsMs = cfg.pauseBetweenRepsMs,
            recitationMode = RecitationMode.SPOKEN
        )
    }

    return PracticeRunSpec(
        practiceId = practice.id,
        title = practice.title,
        defaultPaceWpm = practice.defaultPaceWpm,
        meritDedication = practice.meritDedication,
        segments = mappedSegments,
        mantraConfig = mappedMantra
    )
}

@Composable
fun ParamitaNavGraph(
    viewModel: ParamitaAdvisorViewModel,
    navController: NavHostController = rememberNavController(),
    modifier: Modifier = Modifier
) {
    val liveServiceSnapshot by rememberPracticeServiceState()
    LaunchedEffect(liveServiceSnapshot) {
        viewModel.syncRunnerSnapshot(liveServiceSnapshot)
    }

    val homeState by viewModel.homeUiState.collectAsState()
    val jewelPractices by viewModel.practicesForSelectedJewel.collectAsState()

    NavHost(
        navController = navController,
        startDestination = ParamitaRoutes.HOME,
        modifier = modifier
    ) {
        // 1. Pantalla principal: Refugio / Hoy
        composable(ParamitaRoutes.HOME) {
            RefugeHomeScreen(
                uiState = homeState,
                onOpenPractice = { practiceId ->
                    navController.navigate(ParamitaRoutes.practiceRoute(practiceId))
                },
                onOpenImpulsePause = {
                    navController.navigate(ParamitaRoutes.IMPULSE_PAUSE)
                },
                onOpenThreeJewels = { jewel ->
                    viewModel.selectJewel(jewel)
                    navController.navigate(ParamitaRoutes.threeJewelsRoute(jewel))
                },
                onDedicateTodayMerit = { text ->
                    viewModel.dedicateTodayMerit(text)
                },
                onOpenFaq = {
                    navController.navigate(ParamitaRoutes.FAQ)
                }
            )
        }

        // 2. Pantalla de recitación de oraciones, mantras y meditaciones guiadas
        composable(
            route = ParamitaRoutes.PRACTICE,
            arguments = listOf(navArgument("practiceId") { type = NavType.StringType })
        ) { backStackEntry ->
            val practiceId = backStackEntry.arguments?.getString("practiceId").orEmpty()
            val specFlow = remember(practiceId) { viewModel.observePracticeSpec(practiceId) }
            val spec by specFlow.collectAsState(initial = null)

            spec?.let { loadedSpec ->
                PracticeRunnerScreen(
                    spec = loadedSpec,
                    onMeritDedicatedAndClose = { id, dedication ->
                        viewModel.dedicateMeritFromPractice(id, dedication)
                        navController.popBackStack()
                    },
                    onNavigateBack = {
                        navController.popBackStack()
                    }
                )
            }
        }

        // 3. Pantalla de freno a compras impulsivas y calma digital
        composable(ParamitaRoutes.IMPULSE_PAUSE) {
            ImpulsePauseScreen(
                onSaveMeritAndCalmEvent = { event, decision ->
                    viewModel.recordMeritAndCalmEvent(event, decision)
                },
                onStartGuidedMaitriPractice = {
                    navController.navigate(ParamitaRoutes.practiceRoute("prac_med_amor_bondadoso"))
                },
                onProceedConsciously = {
                    navController.popBackStack()
                },
                onCloseToCalm = {
                    navController.popBackStack(ParamitaRoutes.HOME, inclusive = false)
                }
            )
        }

        // 4. Sección Buddha, Dharma y Sangha conectada a paramita.org
        composable(
            route = ParamitaRoutes.THREE_JEWELS,
            arguments = listOf(
                navArgument("jewel") {
                    type = NavType.StringType
                    defaultValue = JewelTab.DHARMA.key
                }
            )
        ) { backStackEntry ->
            val jewelKey = backStackEntry.arguments?.getString("jewel")
            LaunchedEffect(jewelKey) {
                JewelTab.entries.firstOrNull { it.key == jewelKey }?.let {
                    viewModel.selectJewel(it)
                }
            }

            RefugeAndParamitaScreen(
                practicesForJewel = jewelPractices,
                onOpenPractice = { practiceId ->
                    navController.navigate(ParamitaRoutes.practiceRoute(practiceId))
                },
                onRecordSanghaConnection = { event ->
                    viewModel.recordSanghaConnection(event)
                }
            )
        }

        // 5. Guía y Preguntas Frecuentes (alcance ético, sin reemplazo de maestro ni logros de nirvana)
        composable(ParamitaRoutes.FAQ) {
            FaqScreen(onNavigateBack = { navController.popBackStack() })
        }
    }
}

Implementá el AccessibilityService opt-in en Kotlin que use MindfulAppTriggerEvaluator para lanzar ImpulsePauseActivity.

Vamos a implementar MindfulAppAccessibilityService, un AccessibilityService opt-in en Kotlin que escucha cambios de ventana en primer plano (TYPE_WINDOW_STATE_CHANGED), consulta MindfulAppTriggerEvaluator contra mindful_app_targets respetando cooldown_minutes y lanza ImpulsePauseActivity cuando corresponde, sin leer contenido de pantalla ni retener datos sensibles.

1. Configuración XML del servicio (res/xml/mindful_accessibility_service.xml)
Solo solicitamos TYPE_WINDOW_STATE_CHANGED y un notificationTimeout moderado (250 ms) para no sobrecargar el sistema ni inspeccionar jerarquías de vistas innecesarias.

XML

<accessibility-service xmlns:android="http://schemas.android.com/apk/res/android"
    android:accessibilityEventTypes="typeWindowStateChanged"
    android:accessibilityFeedbackType="feedbackGeneric"
    android:accessibilityFlags="flagDefault"
    android:canRetrieveWindowContent="false"
    android:description="@string/accessibility_service_description"
    android:notificationTimeout="250"
    android:settingsActivity="com.paramita.advisor.MainActivity" />
Y en res/values/strings.xml:

XML

3. Implementación de MindfulAppAccessibilityService en Kotlin
El servicio mantiene su propio CoroutineScope (SupervisorJob() + Dispatchers.Main.immediate), cancela trabajos pendientes al destruirse para evitar fugas, ignora eventos sin packageName o de la propia app, y delega la decisión de enfriamiento (cooldown_minutes) y habilitación (prompt_enabled) a MindfulAppTriggerEvaluator.

Kotlin
package com.paramita.advisor.digitalwellbeing

import android.accessibilityservice.AccessibilityService
import android.content.Context
import android.content.Intent
import android.provider.Settings
import android.view.accessibility.AccessibilityEvent
import com.paramita.advisor.data.local.dao.MindfulAppTargetDao
import com.paramita.advisor.ui.wellbeing.ImpulsePauseActivity
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.launch

class MindfulAppAccessibilityService : AccessibilityService() {

    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)
    private var evaluationJob: Job? = null

    private val cooldownTracker by lazy { ImpulseCooldownTracker() }

    private val targetDao: MindfulAppTargetDao by lazy {
        (application as MindfulTargetDaoHolder).mindfulAppTargetDao
    }

    private val triggerEvaluator: MindfulAppTriggerEvaluator by lazy {
        MindfulAppTriggerEvaluator(
            targetDao = targetDao,
            cooldownTracker = cooldownTracker
        )
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        if (event == null || event.eventType != AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) {
            return
        }

        val incomingPackage = event.packageName?.toString()?.trim()
        if (incomingPackage.isNullOrEmpty() || incomingPackage == packageName) {
            return
        }

        evaluationJob?.cancel()
        evaluationJob = serviceScope.launch {
            val matchedTarget = triggerEvaluator.evaluateForegroundPackage(
                packageName = incomingPackage,
                ownPackageName = packageName
            ) ?: return@launch

            ImpulsePauseActivity.launch(
                context = this@MindfulAppAccessibilityService,
                packageName = matchedTarget.packageName,
                appDisplayName = matchedTarget.displayName
            )
        }
    }

    override fun onInterrupt() {
        evaluationJob?.cancel()
    }

    override fun onDestroy() {
        evaluationJob?.cancel()
        serviceScope.cancel()
        super.onDestroy()
    }

    companion object {
        fun openSystemAccessibilitySettings(context: Context) {
            val intent = Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS).apply {
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }
            context.startActivity(intent)
        }
    }
}

interface MindfulTargetDaoHolder {
    val mindfulAppTargetDao: MindfulAppTargetDao
}
4. Cuidados importantes en Android
Opt-in real del usuario: el sistema nunca habilita un AccessibilityService por código; openSystemAccessibilitySettings lleva al usuario a los Ajustes de Accesibilidad donde él decide activarlo tras leer la descripción clara.

canRetrieveWindowContent="false": refuerza el principio de bienestar digital y privacidad: el servicio solo necesita saber qué paquete pasó al frente (packageName), nunca inspeccionar el contenido de la pantalla.

Evitar rebotes al volver de la pausa: como MindfulAppTriggerEvaluator.evaluateForegroundPackage llama a cooldownTracker.markShown(...) antes de lanzar ImpulsePauseActivity, si el usuario elige "Continuar con consciencia" y regresa a la app de compras o scroll, el cooldown_minutes ya está activo y no se dispara en bucle.

Creá una pantalla en Jetpack Compose para agregar o editar apps en mindful_app_targets y configurar cooldown_minutes y el acceso de accesibilidad.

Vamos a armar MindfulAppTargetsScreen en Jetpack Compose, pensada para que puedas agregar nuevas apps objetivo, editar las existentes (activar/desactivar promptEnabled, ajustar cooldownMinutes y cambiar categoría SHOPPING / SOCIAL_SCROLL / OTHER o eliminarlas) y verificar/activar el AccessibilityService opt-in desde la misma vista.

1. Verificación del estado actual de MindfulAppAccessibilityService
En Android, consultar Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES permite mostrar con claridad si el servicio de accesibilidad ya está activo o si conviene abrir los ajustes del sistema para habilitarlo.

Kotlin
package com.paramita.advisor.digitalwellbeing

import android.content.ComponentName
import android.content.Context
import android.provider.Settings

fun isMindfulAccessibilityServiceEnabled(context: Context): Boolean {
    val expectedComponent = ComponentName(
        context,
        MindfulAppAccessibilityService::class.java
    ).flattenToString()

    val enabledServices = Settings.Secure.getString(
        context.contentResolver,
        Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
    ) ?: return false

    return enabledServices
        .split(':')
        .any { it.equals(expectedComponent, ignoreCase = true) }
}

private fun queryLaunchableApps(
    packageManager: PackageManager,
    ownPackageName: String
): List<InstalledLaunchableApp> {
    val intent = Intent(Intent.ACTION_MAIN, null).apply {
        addCategory(Intent.CATEGORY_LAUNCHER)
    }

    val resolveInfos = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
        packageManager.queryIntentActivities(
            intent,
            PackageManager.ResolveInfoFlags.of(PackageManager.MATCH_DEFAULT_ONLY.toLong())
        )
    } else {
        @Suppress("DEPRECATION")
        packageManager.queryIntentActivities(intent, 0)
    }

    return resolveInfos
        .mapNotNull { info ->
            val pkg = info.activityInfo?.packageName ?: return@mapNotNull null
            if (pkg == ownPackageName) return@mapNotNull null
            val label = info.loadLabel(packageManager)?.toString()?.trim().orEmpty()
            InstalledLaunchableApp(
                displayName = label.ifEmpty { pkg },
                packageName = pkg
            )
        }
        .distinctBy { it.packageName }
        .sortedBy { it.displayName.lowercase() }
}

class ParamitaAdvisorViewModel(
    private val practiceDao: PracticeDao,
    private val practiceSessionDao: PracticeSessionDao,
    private val meritAndCalmDao: MeritAndCalmDao,
    private val mindfulAppTargetDao: MindfulAppTargetDao
) : ViewModel() {

    private val activeRunnerSnapshot = MutableStateFlow(RunnerSnapshot())
    private val selectedJewelTab = MutableStateFlow(JewelTab.DHARMA)

    private val startOfTodayMs: Long
        get() = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, 0)
            set(Calendar.MINUTE, 0)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
        }.timeInMillis

    private val startOfWeekMs: Long
        get() = startOfTodayMs - (6L * 24L * 60L * 60L * 1000L)

    val homeUiState: StateFlow<RefugeHomeUiState> = combine(
        practiceDao.observeAllPracticesWithDetails(),
        practiceSessionDao.observeCompletedSessionsSince(startOfWeekMs),
        meritAndCalmDao.observeEventCountSince("IMPULSE_PURCHASE_AVOIDED", startOfWeekMs),
        meritAndCalmDao.observeEvents(limit = 30),
        activeRunnerSnapshot
    ) { allPractices, completedSessionsWeek, avoidedPurchasesWeek, recentEvents, runnerSnap ->

        val todayEvents = recentEvents.filter { it.createdAtEpochMs >= startOfTodayMs }
        val hasPracticedToday = completedSessionsWeek > 0 &&
            todayEvents.any { it.eventType == "GENEROSITY_MERIT" }
        val hasPausedImpulseToday = todayEvents.any {
            it.eventType == "IMPULSE_PURCHASE_AVOIDED" || it.eventType == "DIGITAL_CALM_BREAK"
        }
        val hasSanghaToday = todayEvents.any { it.eventType == "SANGHA_CONNECTION" }

        val featured = allPractices.firstOrNull { it.practice.id == "prac_med_amor_bondadoso" }
            ?: allPractices.firstOrNull()

        val seeds = listOf(
            DailySeedSummary(
                title = "Semilla de Refugio y Recitación",
                description = "Abrir la práctica con intención clara y acompañar oraciones o mantras.",
                isCultivatedToday = hasPracticedToday
            ),
            DailySeedSummary(
                title = "Mano Abierta y Calma Digital",
                description = "Detener impulsos de compra o scroll automático con amor bondadoso.",
                isCultivatedToday = hasPausedImpulseToday
            ),
            DailySeedSummary(
                title = "Pulso de Sangha",
                description = "Sostener el estudio o la meditación en vínculo con la comunidad.",
                isCultivatedToday = hasSanghaToday
            )
        )

        RefugeHomeUiState(
            featuredPractice = featured,
            quickPractices = allPractices.take(6),
            activeRunnerSnapshot = runnerSnap,
            recentMeritCountThisWeek = completedSessionsWeek + recentEvents.count {
                it.createdAtEpochMs >= startOfWeekMs
            },
            recentImpulsePausesThisWeek = avoidedPurchasesWeek + recentEvents.count {
                it.createdAtEpochMs >= startOfWeekMs && it.eventType == "DIGITAL_CALM_BREAK"
            },
            dailySeeds = seeds
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5_000L),
        initialValue = RefugeHomeUiState()
    )

    val practicesForSelectedJewel: StateFlow<List<PracticeWithDetails>> = combine(
        practiceDao.observeAllPracticesWithDetails(),
        selectedJewelTab
    ) { practices, jewel ->
        practices.filter { jewel.key in it.jewels }
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5_000L),
        initialValue = emptyList()
    )

    val mindfulAppTargets: StateFlow<List<MindfulAppTargetEntity>> = mindfulAppTargetDao
        .observeTargets()
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5_000L),
            initialValue = emptyList()
        )

    fun syncRunnerSnapshot(snapshot: RunnerSnapshot) {
        activeRunnerSnapshot.value = snapshot
    }

    fun selectJewel(jewel: JewelTab) {
        selectedJewelTab.value = jewel
    }

    fun observePracticeSpec(practiceId: String): Flow<PracticeRunSpec?> {
        return practiceDao.observePracticeById(practiceId).map { details ->
            details?.toPracticeRunSpec()
        }
    }

    fun saveMindfulAppTarget(target: MindfulAppTargetEntity) {
        viewModelScope.launch {
            mindfulAppTargetDao.upsertTarget(target)
        }
    }

    fun toggleMindfulAppTargetEnabled(target: MindfulAppTargetEntity, enabled: Boolean) {
        viewModelScope.launch {
            mindfulAppTargetDao.updateTargetRule(
                packageName = target.packageName,
                enabled = enabled,
                cooldownMinutes = target.cooldownMinutes.coerceAtLeast(1)
            )
        }
    }

    fun deleteMindfulAppTarget(packageName: String) {
        viewModelScope.launch {
            mindfulAppTargetDao.deleteTarget(packageName)
        }
    }

    fun recordMeritAndCalmEvent(event: MeritAndCalmEventEntity, decision: ImpulsePauseDecision) {
        viewModelScope.launch {
            meritAndCalmDao.insertEvent(event)
        }
    }

    fun recordSanghaConnection(event: MeritAndCalmEventEntity) {
        viewModelScope.launch {
            meritAndCalmDao.insertEvent(event)
        }
    }

    fun dedicateMeritFromPractice(practiceId: String, dedicationText: String) {
        viewModelScope.launch {
            practiceSessionDao.getLatestSession()?.let { latest ->
                if (latest.practiceId == practiceId) {
                    practiceSessionDao.markMeritDedicated(latest.id)
                }
            }
            meritAndCalmDao.insertEvent(
                MeritAndCalmEventEntity(
                    id = UUID.randomUUID().toString(),
                    eventType = "GENEROSITY_MERIT",
                    title = "Dedicación de mérito tras práctica",
                    triggerContext = practiceId,
                    dedicationNote = dedicationText,
                    createdAtEpochMs = System.currentTimeMillis()
                )
            )
        }
    }

    fun dedicateTodayMerit(dedicationText: String) {
        viewModelScope.launch {
            meritAndCalmDao.insertEvent(
                MeritAndCalmEventEntity(
                    id = UUID.randomUUID().toString(),
                    eventType = "GENEROSITY_MERIT",
                    title = "Dedicación del mérito del día",
                    triggerContext = "Refugio / Hoy",
                    dedicationNote = dedicationText,
                    createdAtEpochMs = System.currentTimeMillis()
                )
            )
        }
    }

    companion object {
        fun factory(
            practiceDao: PracticeDao,
            practiceSessionDao: PracticeSessionDao,
            meritAndCalmDao: MeritAndCalmDao,
            mindfulAppTargetDao: MindfulAppTargetDao
        ): ViewModelProvider.Factory = object : ViewModelProvider.Factory {
            @Suppress("UNCHECKED_CAST")
            override fun <T : ViewModel> create(modelClass: Class<T>): T {
                return ParamitaAdvisorViewModel(
                    practiceDao = practiceDao,
                    practiceSessionDao = practiceSessionDao,
                    meritAndCalmDao = meritAndCalmDao,
                    mindfulAppTargetDao = mindfulAppTargetDao
                ) as T
            }
        }
    }
}

fun PracticeWithDetails.toPracticeRunSpec(): PracticeRunSpec {
    val mappedSegments = orderedSegments.map { seg ->
        PracticeSegmentModel(
            id = seg.id,
            order = seg.segmentOrder,
            phase = seg.phase,
            heading = seg.heading,
            text = seg.text,
            transliteration = seg.transliteration,
            meaning = seg.meaning,
            pauseAfterSec = seg.pauseAfterSec,
            repeatCount = seg.repeatCount
        )
    }

    val mappedMantra = mantraConfig?.let { cfg ->
        MantraRunConfig(
            targetRepetitions = cfg.targetRepetitions,
            bellEveryN = cfg.bellEveryN,
            pauseBetweenRepsMs = cfg.pauseBetweenRepsMs,
            recitationMode = RecitationMode.SPOKEN
        )
    }

    return PracticeRunSpec(
        practiceId = practice.id,
        title = practice.title,
        defaultPaceWpm = practice.defaultPaceWpm,
        meritDedication = practice.meritDedication,
        segments = mappedSegments,
        mantraConfig = mappedMantra
    )
}

object ParamitaRoutes {
    const val HOME = "home"
    const val PRACTICE = "practice/{practiceId}"
    const val IMPULSE_PAUSE = "impulse_pause"
    const val MINDFUL_TARGETS = "mindful_targets"
    const val THREE_JEWELS = "three_jewels?jewel={jewel}"
    const val FAQ = "faq"

    fun practiceRoute(practiceId: String): String = "practice/$practiceId"
    fun threeJewelsRoute(jewel: JewelTab): String = "three_jewels?jewel=${jewel.key}"
}

@Composable
fun ParamitaNavGraph(
    viewModel: ParamitaAdvisorViewModel,
    navController: NavHostController = rememberNavController(),
    modifier: Modifier = Modifier
) {
    val liveServiceSnapshot by rememberPracticeServiceState()
    LaunchedEffect(liveServiceSnapshot) {
        viewModel.syncRunnerSnapshot(liveServiceSnapshot)
    }

    val homeState by viewModel.homeUiState.collectAsState()
    val jewelPractices by viewModel.practicesForSelectedJewel.collectAsState()
    val mindfulTargets by viewModel.mindfulAppTargets.collectAsState()

    NavHost(
        navController = navController,
        startDestination = ParamitaRoutes.HOME,
        modifier = modifier
    ) {
        // 1. Pantalla principal: Refugio / Hoy
        composable(ParamitaRoutes.HOME) {
            RefugeHomeScreen(
                uiState = homeState,
                onOpenPractice = { practiceId ->
                    navController.navigate(ParamitaRoutes.practiceRoute(practiceId))
                },
                onOpenImpulsePause = {
                    navController.navigate(ParamitaRoutes.IMPULSE_PAUSE)
                },
                onOpenThreeJewels = { jewel ->
                    viewModel.selectJewel(jewel)
                    navController.navigate(ParamitaRoutes.threeJewelsRoute(jewel))
                },
                onDedicateTodayMerit = { text ->
                    viewModel.dedicateTodayMerit(text)
                },
                onOpenFaq = {
                    navController.navigate(ParamitaRoutes.FAQ)
                }
            )
        }

        // 2. Pantalla de recitación de oraciones, mantras y meditaciones guiadas
        composable(
            route = ParamitaRoutes.PRACTICE,
            arguments = listOf(navArgument("practiceId") { type = NavType.StringType })
        ) { backStackEntry ->
            val practiceId = backStackEntry.arguments?.getString("practiceId").orEmpty()
            val specFlow = remember(practiceId) { viewModel.observePracticeSpec(practiceId) }
            val spec by specFlow.collectAsState(initial = null)

            spec?.let { loadedSpec ->
                PracticeRunnerScreen(
                    spec = loadedSpec,
                    onMeritDedicatedAndClose = { id, dedication ->
                        viewModel.dedicateMeritFromPractice(id, dedication)
                        navController.popBackStack()
                    },
                    onNavigateBack = {
                        navController.popBackStack()
                    }
                )
            }
        }

        // 3. Pantalla de freno a compras impulsivas y calma digital
        composable(ParamitaRoutes.IMPULSE_PAUSE) {
            ImpulsePauseScreen(
                onSaveMeritAndCalmEvent = { event, decision ->
                    viewModel.recordMeritAndCalmEvent(event, decision)
                },
                onStartGuidedMaitriPractice = {
                    navController.navigate(ParamitaRoutes.practiceRoute("prac_med_amor_bondadoso"))
                },
                onProceedConsciously = {
                    navController.popBackStack()
                },
                onCloseToCalm = {
                    navController.popBackStack(ParamitaRoutes.HOME, inclusive = false)
                }
            )
        }

        // 4. Configuración de apps objetivo, cooldown_minutes y accesibilidad opt-in
        composable(ParamitaRoutes.MINDFUL_TARGETS) {
            MindfulAppTargetsScreen(
                targets = mindfulTargets,
                onSaveTarget = { target ->
                    viewModel.saveMindfulAppTarget(target)
                },
                onDeleteTarget = { pkg ->
                    viewModel.deleteMindfulAppTarget(pkg)
                },
                onToggleTargetEnabled = { target, enabled ->
                    viewModel.toggleMindfulAppTargetEnabled(target, enabled)
                },
                onNavigateBack = {
                    navController.popBackStack()
                }
            )
        }

        // 5. Sección Buddha, Dharma y Sangha conectada a paramita.org
        composable(
            route = ParamitaRoutes.THREE_JEWELS,
            arguments = listOf(
                navArgument("jewel") {
                    type = NavType.StringType
                    defaultValue = JewelTab.DHARMA.key
                }
            )
        ) { backStackEntry ->
            val jewelKey = backStackEntry.arguments?.getString("jewel")
            LaunchedEffect(jewelKey) {
                JewelTab.entries.firstOrNull { it.key == jewelKey }?.let {
                    viewModel.selectJewel(it)
                }
            }

            RefugeAndParamitaScreen(
                practicesForJewel = jewelPractices,
                onOpenPractice = { practiceId ->
                    navController.navigate(ParamitaRoutes.practiceRoute(practiceId))
                },
                onRecordSanghaConnection = { event ->
                    viewModel.recordSanghaConnection(event)
                }
            )
        }

        // 6. Guía y Preguntas Frecuentes
        composable(ParamitaRoutes.FAQ) {
            FaqScreen(onNavigateBack = { navController.popBackStack() })
        }
    }
}

2. Clase ParamitaApplication.kt
Implementa ParamitaApplicationHolder y MindfulTargetDaoHolder para que tanto ImpulsePauseActivity como MindfulAppAccessibilityService accedan a los DAOs sin acoplarse a una actividad concreta, y dispara CatalogSeeder.seedIfNeeded() en segundo plano al iniciar la APK.

Kotlin
package com.paramita.advisor

import android.app.Application
import androidx.room.Room
import com.paramita.advisor.data.catalog.CatalogSeeder
import com.paramita.advisor.data.local.ParamitaDatabase
import com.paramita.advisor.data.local.dao.MeritAndCalmDao
import com.paramita.advisor.data.local.dao.MindfulAppTargetDao
import com.paramita.advisor.data.local.dao.PracticeDao
import com.paramita.advisor.data.local.dao.PracticeSessionDao
import com.paramita.advisor.digitalwellbeing.MindfulTargetDaoHolder
import com.paramita.advisor.ui.wellbeing.ParamitaApplicationHolder
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.launch

class ParamitaApplication : Application(), ParamitaApplicationHolder, MindfulTargetDaoHolder {

    private val applicationScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    val database: ParamitaDatabase by lazy {
        Room.databaseBuilder(
            applicationContext,
            ParamitaDatabase::class.java,
            "paramita_advisor.db"
        )
            .fallbackToDestructiveMigration(dropAllTables = true)
            .build()
    }

    val practiceDao: PracticeDao by lazy { database.practiceDao() }
    val practiceSessionDao: PracticeSessionDao by lazy { database.practiceSessionDao() }

    override val meritAndCalmDao: MeritAndCalmDao by lazy { database.meritAndCalmDao() }
    override val mindfulAppTargetDao: MindfulAppTargetDao by lazy { database.mindfulAppTargetDao() }

    val catalogSeeder: CatalogSeeder by lazy {
        CatalogSeeder(
            context = applicationContext,
            practiceDao = practiceDao
        )
    }

    override fun onCreate() {
        super.onCreate()
        applicationScope.launch {
            catalogSeeder.seedIfNeeded()
        }
    }
}

2. Implementación de SpeechAndBellController con SoundPool
Kotlin
package com.paramita.advisor.engine

import android.content.Context
import android.media.AudioAttributes
import android.media.ToneGenerator
import android.media.SoundPool
import android.os.Handler
import android.os.Looper
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import kotlinx.coroutines.suspendCancellableCoroutine
import java.util.Locale
import java.util.UUID
import kotlin.coroutines.resume

class SpeechAndBellController(
    private val context: Context
) {
    private val appContext = context.applicationContext
    private val mainHandler = Handler(Looper.getMainLooper())

    private var tts: TextToSpeech? = null
    @Volatile
    private var isTtsReady = false

    private var bellSoundId: Int = 0
    @Volatile
    private var isBellLoaded = false
    private var pendingBellPlay = false

    private val audioAttributes = AudioAttributes.Builder()
        .setUsage(AudioAttributes.USAGE_MEDIA)
        .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
        .build()

    private val soundPool: SoundPool = SoundPool.Builder()
        .setMaxStreams(2)
        .setAudioAttributes(audioAttributes)
        .build().apply {
            setOnLoadCompleteListener { _, sampleId, status ->
                if (status == 0 && sampleId == bellSoundId) {
                    isBellLoaded = true
                    if (pendingBellPlay) {
                        pendingBellPlay = false
                        dispatchBellStream()
                    }
                }
            }
        }

    init {
        initBellResource()
        initTextToSpeech()
    }

    private fun initBellResource() {
        val resId = appContext.resources.getIdentifier(
            "bell_soft",
            "raw",
            appContext.packageName
        )
        if (resId != 0) {
            bellSoundId = soundPool.load(appContext, resId, 1)
        }
    }

    private fun initTextToSpeech() {
        tts = TextToSpeech(appContext) { status ->
            if (status == TextToSpeech.SUCCESS) {
                val spanishLocale = Locale("es", "AR")
                val fallbackSpanish = Locale("es")
                val result = tts?.setLanguage(spanishLocale)
                if (result == TextToSpeech.LANG_MISSING_DATA || result == TextToSpeech.LANG_NOT_SUPPORTED) {
                    tts?.setLanguage(fallbackSpanish)
                }
                isTtsReady = true
            }
        }
    }

    fun playBell(volumeScale: Float = 0.65f) {
        val clampedVolume = volumeScale.coerceIn(0.1f, 1.0f)

        if (bellSoundId != 0) {
            if (isBellLoaded) {
                soundPool.play(
                    bellSoundId,
                    clampedVolume,
                    clampedVolume,
                    1,
                    0,
                    1.0f
                )
            } else {
                pendingBellPlay = true
            }
            return
        }

        // Respaldo suave si aún no se agregó res/raw/bell_soft
        mainHandler.post {
            runCatching {
                val tone = ToneGenerator(
                    android.media.AudioManager.STREAM_MUSIC,
                    (clampedVolume * 65).toInt()
                )
                tone.startTone(ToneGenerator.TONE_PROP_BEEP, 280)
                mainHandler.postDelayed({ tone.release() }, 350L)
            }
        }
    }

    private fun dispatchBellStream() {
        if (bellSoundId != 0 && isBellLoaded) {
            soundPool.play(bellSoundId, 0.65f, 0.65f, 1, 0, 1.0f)
        }
    }

    suspend fun speak(
        text: String,
        paceWpm: Int = 120,
        mode: RecitationMode = RecitationMode.SPOKEN
    ): Unit = suspendCancellableCoroutine { continuation ->
        if (text.isBlank() || mode == RecitationMode.MENTAL) {
            if (continuation.isActive) continuation.resume(Unit)
            return@suspendCancellableCoroutine
        }

        val engine = tts
        if (!isTtsReady || engine == null) {
            if (continuation.isActive) continuation.resume(Unit)
            return@suspendCancellableCoroutine
        }

        val utteranceId = "utt_${UUID.randomUUID()}"
        val speechRate = (paceWpm / 120f).coerceIn(0.65f, 1.45f)
        engine.setSpeechRate(speechRate)

        engine.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(id: String?) = Unit

            override fun onDone(id: String?) {
                if (id == utteranceId && continuation.isActive) {
                    continuation.resume(Unit)
                }
            }

            @Deprecated("Deprecated in Java")
            override fun onError(id: String?) {
                if (id == utteranceId && continuation.isActive) {
                    continuation.resume(Unit)
                }
            }

            override fun onError(id: String?, errorCode: Int) {
                if (id == utteranceId && continuation.isActive) {
                    continuation.resume(Unit)
                }
            }
        })

        continuation.invokeOnCancellation {
            if (engine.isSpeaking) {
                engine.stop()
            }
        }

        val speakResult = engine.speak(text, TextToSpeech.QUEUE_FLUSH, null, utteranceId)
        if (speakResult == TextToSpeech.ERROR && continuation.isActive) {
            continuation.resume(Unit)
        }
    }

    fun stopSpeaking() {
        pendingBellPlay = false
        if (isTtsReady) {
            tts?.stop()
        }
    }

    fun release() {
        pendingBellPlay = false
        stopSpeaking()
        tts?.shutdown()
        tts = null
        soundPool.release()
    }
}
3. Por qué SoundPool encaja bien aquí
Disparo inmediato en ciclos de repetición: en recitaciones de mantras (cada bell_every_n vueltas o al cerrar un segmento), SoundPool evita la latencia de preparar un MediaPlayer en cada campanada.

Respeto a RecitationMode.MENTAL: cuando la recitación pasa a modo mental/silencioso, speak(...) retorna de inmediato sin emitir voz ni interrumpir el silencio, mientras la campana sigue marcando el ritmo si está configurada.

2. Filtro y mapeo seguro (DrivingRecitationCatalogMapper)
Este componente filtra estrictamente por PRAYER y MANTRA, exige que ttsEnabled esté activo o haya texto recitable, excluye cualquier GUIDED_MEDITATION o IMPULSE_PAUSE (aunque tengan etiquetas de calma), y construye un spokenText conciso a partir de los segmentos ordenados.

Kotlin
package com.paramita.advisor.auto

import com.paramita.advisor.data.local.entity.PracticeWithDetails

class DrivingRecitationCatalogMapper {

    fun filterAndMapForDriving(
        practices: List<PracticeWithDetails>
    ): List<AutoRecitationItem> {
        return practices
            .mapNotNull { details -> mapIfSafeForDriving(details) }
            .sortedWith(
                compareBy<AutoRecitationItem> { it.kind.ordinal }
                    .thenBy { it.title.lowercase() }
            )
    }

    fun mapIfSafeForDriving(details: PracticeWithDetails): AutoRecitationItem? {
        val practice = details.practice
        val kind = when (practice.type.trim().uppercase()) {
            DrivingPracticeKind.PRAYER.wireType -> DrivingPracticeKind.PRAYER
            DrivingPracticeKind.MANTRA.wireType -> DrivingPracticeKind.MANTRA
            else -> return null // Excluye explícitamente GUIDED_MEDITATION, IMPULSE_PAUSE y otros
        }

        if (!practice.ttsEnabled) return null

        val ordered = details.orderedSegments
        val consolidatedText = ordered
            .map { seg -> seg.transliteration?.takeIf { it.isNotBlank() } ?: seg.text.trim() }
            .filter { it.isNotBlank() }
            .joinToString(separator = ". ")
            .trim()

        if (consolidatedText.isEmpty()) return null

        val mantraCfg = details.mantraConfig
        val repetitions = when (kind) {
            DrivingPracticeKind.MANTRA -> (mantraCfg?.targetRepetitions ?: 7).coerceIn(1, 108)
            DrivingPracticeKind.PRAYER -> (ordered.firstOrNull()?.repeatCount ?: 1).coerceIn(1, 7)
        }

        val bellInterval = when (kind) {
            DrivingPracticeKind.MANTRA -> (mantraCfg?.bellEveryN ?: 7).coerceAtLeast(0)
            DrivingPracticeKind.PRAYER -> 0
        }

        val pauseMs = when (kind) {
            DrivingPracticeKind.MANTRA -> (mantraCfg?.pauseBetweenRepsMs ?: 900L).coerceIn(500L, 2500L)
            DrivingPracticeKind.PRAYER -> 1000L
        }

        val safePace = practice.defaultPaceWpm.coerceIn(95, 145)
        val subtitle = practice.subtitle?.takeIf { it.isNotBlank() }
            ?: "${kind.labelEs} · ${repetitions} ${if (repetitions == 1) "vez" else "vueltas"}"

        return AutoRecitationItem(
            practiceId = practice.id,
            kind = kind,
            title = practice.title.trim(),
            shortSubtitle = subtitle,
            spokenText = consolidatedText,
            paceWpm = safePace,
            targetRepetitions = repetitions,
            bellEveryN = bellInterval,
            pauseBetweenRepsMs = pauseMs,
            meritDedication = practice.meritDedication.trim()
        )
    }
}
3. Reglas clave de este diseño para el auto
Exclusión estructural: aunque en el futuro se agreguen meditaciones de 37 Prácticas, maitri o pausas de impulso al catálogo JSON, mapIfSafeForDriving solo deja pasar PRAYER y MANTRA.

Ritmo adaptado a conducción: acota paceWpm entre 95 y 145 y pauseBetweenRepsMs entre 500 y 2500 ms para que la recitación acompañe sin generar silencios largos que inviten a desconectarse del tránsito ni velocidades atropelladas.

1. Puente entre el servicio, la base de datos y las pantallas del auto (ServiceBackedAutoRecitationController)
Este controlador toma los ítems filtrados desde Room (PracticeDao), envía las órdenes a PracticePlaybackService y traduce el RunnerSnapshot del servicio en un AutoPlaybackState limpio para DrivingSelectionScreen y DrivingPlaybackScreen.

Kotlin
package com.paramita.advisor.auto

import android.content.Context
import android.content.Intent
import androidx.car.app.ScreenManager
import androidx.core.content.ContextCompat
import com.paramita.advisor.data.local.dao.PracticeDao
import com.paramita.advisor.engine.RunnerSnapshot
import com.paramita.advisor.engine.RunnerStatus
import com.paramita.advisor.service.PracticePlaybackService
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.launch

class ServiceBackedAutoRecitationController(
    context: Context,
    private val practiceDao: PracticeDao,
    private val screenManagerProvider: () -> ScreenManager?
) : AutoRecitationController {

    private val appContext = context.applicationContext
    private val mapper = DrivingRecitationCatalogMapper()
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)

    private val availableItems = MutableStateFlow<List<AutoRecitationItem>>(emptyList())
    private val playbackStateFlow = MutableStateFlow(AutoPlaybackState())

    private var lastInvalidatedSignature: String? = null
    private var observationJob: Job? = null

    init {
        startObserving()
    }

    private fun startObserving() {
        observationJob?.cancel()
        observationJob = scope.launch {
            combine(
                practiceDao.observeAllPracticesWithDetails(),
                PracticePlaybackService.sessionState
            ) { allPractices, runnerSnapshot ->
                val safeItems = mapper.filterAndMapForDriving(allPractices)
                availableItems.value = safeItems
                mapToAutoPlaybackState(runnerSnapshot, safeItems)
            }.collectLatest { mappedState ->
                playbackStateFlow.value = mappedState
                invalidateActiveScreenIfNeeded(mappedState)
            }
        }
    }

    override fun currentItems(): List<AutoRecitationItem> = availableItems.value

    override fun playbackState(): AutoPlaybackState = playbackStateFlow.value

    override fun startItem(item: AutoRecitationItem) {
        // Doble resguardo: jamás iniciar si el ítem no es PRAYER o MANTRA
        if (item.kind != DrivingPracticeKind.PRAYER && item.kind != DrivingPracticeKind.MANTRA) {
            return
        }

        val intent = Intent(appContext, PracticePlaybackService::class.java).apply {
            action = PracticePlaybackService.ACTION_START_PRACTICE
            putExtra(PracticePlaybackService.EXTRA_PRACTICE_ID, item.practiceId)
            putExtra(PracticePlaybackService.EXTRA_TARGET_REPS, item.targetRepetitions)
            putExtra(PracticePlaybackService.EXTRA_PACE_WPM, item.paceWpm)
            putExtra(PracticePlaybackService.EXTRA_PAUSE_BETWEEN_REPS_MS, item.pauseBetweenRepsMs)
            putExtra(PracticePlaybackService.EXTRA_BELL_EVERY_N, item.bellEveryN)
        }
        ContextCompat.startForegroundService(appContext, intent)
    }

    override fun togglePauseResume() {
        val current = playbackStateFlow.value
        val actionName = when (current.status) {
            AutoPlaybackStatus.PLAYING -> PracticePlaybackService.ACTION_PAUSE
            AutoPlaybackStatus.PAUSED -> PracticePlaybackService.ACTION_RESUME
            AutoPlaybackStatus.IDLE -> {
                val fallbackItem = current.currentItem ?: availableItems.value.firstOrNull()
                if (fallbackItem != null) {
                    startItem(fallbackItem)
                }
                return
            }
        }

        val intent = Intent(appContext, PracticePlaybackService::class.java).apply {
            action = actionName
        }
        ContextCompat.startForegroundService(appContext, intent)
    }

    override fun stopPlayback() {
        val intent = Intent(appContext, PracticePlaybackService::class.java).apply {
            action = PracticePlaybackService.ACTION_STOP
        }
        ContextCompat.startForegroundService(appContext, intent)
    }

    fun release() {
        observationJob?.cancel()
        scope.cancel()
    }

    private fun mapToAutoPlaybackState(
        snapshot: RunnerSnapshot,
        safeItems: List<AutoRecitationItem>
    ): AutoPlaybackState {
        val matchedItem = snapshot.practiceId?.let { id ->
            safeItems.firstOrNull { it.practiceId == id }
        }

        val status = when (snapshot.status) {
            RunnerStatus.PLAYING -> AutoPlaybackStatus.PLAYING
            RunnerStatus.PAUSED -> AutoPlaybackStatus.PAUSED
            RunnerStatus.IDLE, RunnerStatus.COMPLETED -> AutoPlaybackStatus.IDLE
        }

        val currentRep = if (status == AutoPlaybackStatus.IDLE) {
            0
        } else {
            snapshot.currentRepetition.coerceAtLeast(1)
        }

        val totalReps = matchedItem?.targetRepetitions
            ?: snapshot.totalRepetitions.coerceAtLeast(0)

        return AutoPlaybackState(
            status = status,
            currentItem = matchedItem,
            currentRepetition = currentRep,
            totalRepetitions = totalReps
        )
    }

    private fun invalidateActiveScreenIfNeeded(state: AutoPlaybackState) {
        val signature = "${state.currentItem?.practiceId}|${state.status}|${state.currentRepetition}/${state.totalRepetitions}"
        if (signature == lastInvalidatedSignature) return
        lastInvalidatedSignature = signature

        runCatching {
            screenManagerProvider()?.top?.invalidate()
        }
    }
}

package com.paramita.advisor.service

import androidx.core.app.NotificationCompat
import com.paramita.advisor.MainActivity

import com.paramita.advisor.R
import com.paramita.advisor.engine.PracticeRunnerEngine
import com.paramita.advisor.engine.RunnerSnapshot
import com.paramita.advisor.engine.RunnerStatus
import com.paramita.advisor.engine.SpeechAndBellController
import com.paramita.advisor.ui.toPracticeRunSpec
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.launch

class PracticePlaybackService : Service() {

    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)
    private var startPracticeJob: Job? = null
    private var snapshotCollectorJob: Job? = null

    private val speechAndBell: SpeechAndBellController by lazy {
        SpeechAndBellController(applicationContext)
    }

    private val runnerEngine: PracticeRunnerEngine by lazy {
        PracticeRunnerEngine(
            speechAndBell = speechAndBell,
            scope = serviceScope
        )
    }

    override fun onCreate() {
        super.onCreate()
        ensureNotificationChannel()

        snapshotCollectorJob = serviceScope.launch {
            runnerEngine.snapshot.collectLatest { snap ->
                _sessionState.value = snap
                if (snap.status == RunnerStatus.PLAYING || snap.status == RunnerStatus.PAUSED) {
                    updateForegroundNotification(snap)
                } else if (snap.status == RunnerStatus.IDLE || snap.status == RunnerStatus.COMPLETED) {
                    stopForeground(STOP_FOREGROUND_REMOVE)
                }
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START_PRACTICE -> {
                val practiceId = intent.getStringExtra(EXTRA_PRACTICE_ID).orEmpty()
                if (practiceId.isBlank()) return START_NOT_STICKY

                val targetReps = intent.getIntExtra(EXTRA_TARGET_REPS, -1).takeIf { it > 0 }
                val paceWpm = intent.getIntExtra(EXTRA_PACE_WPM, -1).takeIf { it > 0 }
                val pauseBetweenRepsMs = intent.getLongExtra(EXTRA_PAUSE_BETWEEN_REPS_MS, -1L)
                    .takeIf { it >= 0L }
                val bellEveryN = intent.getIntExtra(EXTRA_BELL_EVERY_N, -1).takeIf { it >= 0 }

                startForeground(
                    NOTIFICATION_ID,
                    buildNotification(
                        title = "Preparando práctica",
                        body = "Iniciando recitación o guía…"
                    )
                )

                startPracticeJob?.cancel()
                startPracticeJob = serviceScope.launch {
                    val app = applicationContext as ParamitaApplication
                    val details = app.practiceDao.getPracticeByIdOnce(practiceId) ?: run {
                        stopForeground(STOP_FOREGROUND_REMOVE)
                        stopSelf()
                        return@launch
                    }

                    val baseSpec = details.toPracticeRunSpec()
                    val adjustedSpec = baseSpec.copy(
                        defaultPaceWpm = paceWpm ?: baseSpec.defaultPaceWpm,
                        mantraConfig = baseSpec.mantraConfig?.copy(
                            targetRepetitions = targetReps ?: baseSpec.mantraConfig.targetRepetitions,
                            pauseBetweenRepsMs = pauseBetweenRepsMs ?: baseSpec.mantraConfig.pauseBetweenRepsMs,
                            bellEveryN = bellEveryN ?: baseSpec.mantraConfig.bellEveryN
                        )
                    )

                    runnerEngine.start(
                        spec = adjustedSpec,
                        overrideRepetitions = targetReps
                    )
                }
            }

            ACTION_PAUSE -> {
                runnerEngine.pause()
            }

            ACTION_RESUME -> {
                runnerEngine.resume()
            }

            ACTION_STOP -> {
                startPracticeJob?.cancel()
                runnerEngine.stop()
                stopForeground(STOP_FOREGROUND_REMOVE)
                stopSelf()
            }
        }

        return START_NOT_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        startPracticeJob?.cancel()
        snapshotCollectorJob?.cancel()
        runnerEngine.release()
        speechAndBell.release()
        serviceScope.cancel()
        super.onDestroy()
    }

    private fun updateForegroundNotification(snapshot: RunnerSnapshot) {
        val title = snapshot.practiceTitle ?: "Práctica en curso"
        val body = if (snapshot.totalRepetitions > 0) {
            val rep = snapshot.currentRepetition.coerceAtLeast(1)
            "Repetición $rep de ${snapshot.totalRepetitions}"
        } else {
            snapshot.currentHeading ?: "Acompañando práctica"
        }
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        nm.notify(NOTIFICATION_ID, buildNotification(title, body))
    }

    private fun buildNotification(title: String, body: String): Notification {
        val openAppIntent = Intent(this, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }
        val contentPendingIntent = PendingIntent.getActivity(
            this,
            0,
            openAppIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val stopIntent = Intent(this, PracticePlaybackService::class.java).apply {
            action = ACTION_STOP
        }
        val stopPendingIntent = PendingIntent.getService(
            this,
            1,
            stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val smallIcon = resources.getIdentifier("ic_lotus_calm", "drawable", packageName)
            .takeIf { it != 0 } ?: android.R.drawable.ic_media_play

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(smallIcon)
            .setContentTitle(title)
            .setContentText(body)
            .setContentIntent(contentPendingIntent)
            .setOngoing(true)
            .setOnlyAlertOnce(true)
            .addAction(0, "Detener", stopPendingIntent)
            .build()
    }

    private fun ensureNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Práctica y recitación",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "Acompañamiento de oraciones, mantras y meditaciones en segundo plano"
            }
            val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            nm.createNotificationChannel(channel)
        }
    }

Para que PracticePlaybackService pueda resolver una práctica por ID en una única consulta suspend (sin abrir un Flow continuo en el servicio), sumamos getPracticeByIdOnce a PracticeDao usando @Transaction para traer la práctica junto con sus segmentos ordenados y su configuración de mantra (PracticeWithDetails).

1. Método getPracticeByIdOnce en PracticeDao

package com.paramita.advisor.data.local.dao

import androidx.room.Dao
import androidx.room.Query
import androidx.room.Transaction
import com.paramita.advisor.data.local.entity.PracticeWithDetails
import kotlinx.coroutines.flow.Flow

@Dao
interface PracticeDao {

    @Transaction
    @Query("SELECT * FROM practices WHERE id = :practiceId LIMIT 1")
    fun observePracticeById(practiceId: String): Flow<PracticeWithDetails?>

    @Transaction
    @Query("SELECT * FROM practices WHERE id = :practiceId LIMIT 1")
    suspend fun getPracticeByIdOnce(practiceId: String): PracticeWithDetails?

    @Transaction
    @Query("SELECT * FROM practices ORDER BY title ASC")
    fun observeAllPracticesWithDetails(): Flow<List<PracticeWithDetails>>

    // ... resto de consultas existentes de prácticas, segmentos y mantras
}
2. Por qué lleva @Transaction
PracticeWithDetails combina la fila de practices con relaciones (@Relation) hacia los segmentos y la configuración de repeticiones/campana del mantra.

En Room, marcar la consulta con @Transaction garantiza que todas las lecturas relacionadas se ejecuten de forma coherente y evita advertencias o inconsistencias al armar el PracticeRunSpec dentro de PracticePlaybackService.
